  ticket.comments = [...ticket.comments, comment];
  tickets = tickets.map(t => t.id === id ? { ...ticket, updated: new Date().toISOString() } : t);
  return getTicket(id);
}

import React, { useState } from 'react'
import TicketList from './components/TicketList'
import TicketDetail from './components/TicketDetail'
import CreateTicketModal from './components/CreateTicketModal'
import { getTickets, createTicket, updateTicket, addComment } from './data/tickets'
import styles from './App.module.css'

export default function App() {
  const [tickets, setTickets] = useState(getTickets)
  const [selectedId, setSelectedId] = useState(null)
  const [showCreate, setShowCreate] = useState(false)

  const refresh = () => setTickets(getTickets())

  const handleCreate = (data) => {
    createTicket(data)
    refresh()
    setShowCreate(false)
  }

  const handleStatusChange = (id, status) => {
    updateTicket(id, { status })
    refresh()
  }

  const handleComment = (id, author, body) => {
    addComment(id, author, body)
    refresh()
  }

  const selected = tickets.find(t => t.id === selectedId) || null

  return (
    <div className={styles.app}>
      <header className={styles.header}>
        <div className={styles.headerInner}>
          <div className={styles.brand}>
            <span className={styles.brandIcon}>🎫</span>
            <span className={styles.brandName}>SupportDesk</span>
          </div>
          <button className={styles.createBtn} onClick={() => setShowCreate(true)}>
            + New Ticket
          </button>
        </div>
      </header>

      <main className={styles.main}>
        <div className={styles.layout}>
          <aside className={styles.sidebar}>
            <TicketList
              tickets={tickets}
              selectedId={selectedId}
              onSelect={setSelectedId}
            />
          </aside>
          <section className={styles.detail}>
            {selected ? (
              <TicketDetail
                ticket={selected}
                onStatusChange={handleStatusChange}
                onComment={handleComment}
              />
            ) : (
              <div className={styles.empty}>
                <div className={styles.emptyIcon}>📋</div>
                <p className={styles.emptyTitle}>No ticket selected</p>
                <p className={styles.emptyHint}>Choose a ticket from the list or create a new one.</p>
              </div>
            )}
          </section>
        </div>
      </main>

      {showCreate && (
        <CreateTicketModal
          onSubmit={handleCreate}
          onClose={() => setShowCreate(false)}
        />
      )}
    </div>
  )
}

import React, { useState, useMemo } from 'react'
import { STATUSES } from '../data/tickets'
import TicketRow from './TicketRow'
import styles from './TicketList.module.css'

const ALL = 'All'
const STATUS_TABS = [ALL, ...STATUSES]

export default function TicketList({ tickets, selectedId, onSelect }) {
  const [search, setSearch] = useState('')
  const [activeStatus, setActiveStatus] = useState(ALL)

  const counts = useMemo(() => {
    const c = { All: tickets.length }
    STATUSES.forEach(s => { c[s] = tickets.filter(t => t.status === s).length })
    return c
  }, [tickets])

  const filtered = useMemo(() => {
    return tickets.filter(t => {
      const matchStatus = activeStatus === ALL || t.status === activeStatus
      const q = search.toLowerCase()
      const matchSearch = !q ||
        t.title.toLowerCase().includes(q) ||
        t.requester.toLowerCase().includes(q) ||
        String(t.id).includes(q) ||
        t.category.toLowerCase().includes(q)
      return matchStatus && matchSearch
    })
  }, [tickets, search, activeStatus])

  return (
    <div className={styles.container}>
      <div className={styles.top}>
        <h2 className={styles.heading}>Tickets</h2>
        <span className={styles.total}>{tickets.length} total</span>
      </div>

      <div className={styles.searchWrap}>
        <span className={styles.searchIcon}>🔍</span>
        <input
          className={styles.search}
          type="text"
          placeholder="Search by title, requester, ID…"
          value={search}
          onChange={e => setSearch(e.target.value)}
        />
        {search && (
          <button className={styles.clearBtn} onClick={() => setSearch('')}>✕</button>
        )}
      </div>

      <div className={styles.tabs}>
        {STATUS_TABS.map(s => (
          <button
            key={s}
            className={`${styles.tab} ${activeStatus === s ? styles.tabActive : ''}`}
            onClick={() => setActiveStatus(s)}
          >
            {s}
            <span className={styles.tabCount}>{counts[s] ?? 0}</span>
          </button>
        ))}
      </div>

      <div className={styles.list}>
        {filtered.length === 0 ? (
          <div className={styles.noResults}>
            <span>😶</span>
            <p>No tickets match your filters.</p>
          </div>
        ) : (
          filtered.map(t => (
            <TicketRow
              key={t.id}
              ticket={t}
              selected={t.id === selectedId}
              onClick={() => onSelect(t.id)}
            />
          ))
        )}
      </div>
    </div>
  )
}

import React, { useState } from 'react'
import { STATUSES, PRIORITIES, CATEGORIES } from '../data/tickets'
import styles from './CreateTicketModal.module.css'

export default function CreateTicketModal({ onSubmit, onClose }) {
  const [form, setForm] = useState({
    title: '',
    requester: '',
    category: CATEGORIES[0],
    priority: 'Medium',
    status: 'Open',
    description: '',
  })
  const [errors, setErrors] = useState({})

  const set = (k, v) => setForm(f => ({ ...f, [k]: v }))

  const validate = () => {
    const e = {}
    if (!form.title.trim()) e.title = 'Title is required'
    if (!form.requester.trim()) e.requester = 'Requester email is required'
    else if (!/\S+@\S+\.\S+/.test(form.requester)) e.requester = 'Enter a valid email'
    if (!form.description.trim()) e.description = 'Description is required'
    return e
  }

  const handleSubmit = (e) => {
    e.preventDefault()
    const e2 = validate()
    if (Object.keys(e2).length) { setErrors(e2); return }
    onSubmit(form)
  }

  return (
    <div className={styles.overlay} onClick={e => e.target === e.currentTarget && onClose()}>
      <div className={styles.modal}>
        <div className={styles.modalHeader}>
          <h2 className={styles.modalTitle}>Create New Ticket</h2>
          <button className={styles.closeBtn} onClick={onClose}>✕</button>
        </div>

        <form className={styles.form} onSubmit={handleSubmit} noValidate>
          <div className={styles.field}>
            <label className={styles.label}>Title <span className={styles.req}>*</span></label>
            <input
              className={`${styles.input} ${errors.title ? styles.inputError : ''}`}
              value={form.title}
              onChange={e => set('title', e.target.value)}
              placeholder="Brief summary of the issue"
            />
            {errors.title && <span className={styles.error}>{errors.title}</span>}
          </div>

          <div className={styles.field}>
            <label className={styles.label}>Requester Email <span className={styles.req}>*</span></label>
            <input
              className={`${styles.input} ${errors.requester ? styles.inputError : ''}`}
              type="email"
              value={form.requester}
              onChange={e => set('requester', e.target.value)}
              placeholder="customer@example.com"
            />
            {errors.requester && <span className={styles.error}>{errors.requester}</span>}
          </div>

          <div className={styles.row}>
            <div className={styles.field}>
              <label className={styles.label}>Category</label>
              <select className={styles.select} value={form.category} onChange={e => set('category', e.target.value)}>
                {CATEGORIES.map(c => <option key={c}>{c}</option>)}
              </select>
            </div>
            <div className={styles.field}>
              <label className={styles.label}>Priority</label>
              <select className={styles.select} value={form.priority} onChange={e => set('priority', e.target.value)}>
                {PRIORITIES.map(p => <option key={p}>{p}</option>)}
              </select>
            </div>
            <div className={styles.field}>
              <label className={styles.label}>Status</label>
              <select className={styles.select} value={form.status} onChange={e => set('status', e.target.value)}>
                {STATUSES.map(s => <option key={s}>{s}</option>)}
              </select>
            </div>
          </div>

          <div className={styles.field}>
            <label className={styles.label}>Description <span className={styles.req}>*</span></label>
            <textarea
              className={`${styles.textarea} ${errors.description ? styles.inputError : ''}`}
              rows={5}
              value={form.description}
              onChange={e => set('description', e.target.value)}
              placeholder="Describe the issue in detail…"
            />
            {errors.description && <span className={styles.error}>{errors.description}</span>}
          </div>

          <div className={styles.actions}>
            <button type="button" className={styles.cancelBtn} onClick={onClose}>Cancel</button>
            <button type="submit" className={styles.submitBtn}>Create Ticket</button>
          </div>
        </form>
      </div>
    </div>
  )
}

export function priorityColor(priority) {
  switch (priority) {
    case 'Critical': return '#ef4444'
    case 'High':     return '#f97316'
    case 'Medium':   return '#f59e0b'
    case 'Low':      return '#22c55e'
    default:         return '#64748b'
  }
}

export function statusColor(status) {
  switch (status) {
    case 'Open':        return { background: '#dbeafe', color: '#1d4ed8' }
    case 'In Progress': return { background: '#fef3c7', color: '#92400e' }
    case 'Resolved':    return { background: '#dcfce7', color: '#166534' }
    case 'Closed':      return { background: '#f1f5f9', color: '#475569' }
    default:            return { background: '#f1f5f9', color: '#475569' }
  }
}

export function formatDate(iso) {
  if (!iso) return ''
  const d = new Date(iso)
  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
}

export function formatDateTime(iso) {
  if (!iso) return ''
  const d = new Date(iso)
  return d.toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
}
