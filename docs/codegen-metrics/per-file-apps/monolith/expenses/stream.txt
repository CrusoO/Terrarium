<<<FILE path="src/hooks/useExpenses.js">>>
import { useState, useMemo } from 'react'
import { INITIAL_EXPENSES } from '../data/initialExpenses'

let nextId = 200

export function useExpenses() {
  const [expenses, setExpenses] = useState(INITIAL_EXPENSES)

  const addExpense = (expense) => {
    setExpenses(prev => [{ ...expense, id: nextId++ }, ...prev])
  }

  const deleteExpense = (id) => {
    setExpenses(prev => prev.filter(e => e.id !== id))
  }

  const totals = useMemo(() => {
    const byCategory = {}
    let grand = 0
    expenses.forEach(e => {
      byCategory[e.category] = (byCategory[e.category] || 0) + e.amount
      grand += e.amount
    })
    return { byCategory, grand }
  }, [expenses])

  return { expenses, addExpense, deleteExpense, totals }
}

import React, { useState, useMemo } from 'react'
import { useExpenses } from './hooks/useExpenses'
import { CATEGORIES, STAFF } from './data/initialExpenses'
import Header from './components/Header'
import Totals from './components/Totals'
import Filters from './components/Filters'
import ExpenseTable from './components/ExpenseTable'
import AddExpenseModal from './components/AddExpenseModal'
import styles from './App.module.css'

export default function App() {
  const { expenses, addExpense, deleteExpense, totals } = useExpenses()
  const [search, setSearch] = useState('')
  const [filterCategory, setFilterCategory] = useState('all')
  const [filterStaff, setFilterStaff] = useState('all')
  const [showModal, setShowModal] = useState(false)
  const [sortField, setSortField] = useState('date')
  const [sortDir, setSortDir] = useState('desc')

  const filtered = useMemo(() => {
    let list = expenses
    if (filterCategory !== 'all') list = list.filter(e => e.category === filterCategory)
    if (filterStaff !== 'all') list = list.filter(e => e.staff === filterStaff)
    if (search.trim()) {
      const q = search.toLowerCase()
      list = list.filter(e =>
        e.description.toLowerCase().includes(q) ||
        e.staff.toLowerCase().includes(q) ||
        e.category.toLowerCase().includes(q)
      )
    }
    list = [...list].sort((a, b) => {
      let va = a[sortField], vb = b[sortField]
      if (sortField === 'amount') { va = +va; vb = +vb }
      if (va < vb) return sortDir === 'asc' ? -1 : 1
      if (va > vb) return sortDir === 'asc' ? 1 : -1
      return 0
    })
    return list
  }, [expenses, filterCategory, filterStaff, search, sortField, sortDir])

  const filteredTotal = useMemo(() => filtered.reduce((s, e) => s + e.amount, 0), [filtered])

  const handleSort = (field) => {
    if (sortField === field) setSortDir(d => d === 'asc' ? 'desc' : 'asc')
    else { setSortField(field); setSortDir('asc') }
  }

  return (
    <div className={styles.app}>
      <Header onAdd={() => setShowModal(true)} />
      <main className={styles.main}>
        <Totals totals={totals} categories={CATEGORIES} />
        <Filters
          search={search} onSearch={setSearch}
          filterCategory={filterCategory} onCategory={setFilterCategory}
          filterStaff={filterStaff} onStaff={setFilterStaff}
          categories={CATEGORIES} staff={STAFF}
        />
        <ExpenseTable
          expenses={filtered}
          categories={CATEGORIES}
          filteredTotal={filteredTotal}
          onDelete={deleteExpense}
          sortField={sortField}
          sortDir={sortDir}
          onSort={handleSort}
        />
      </main>
      {showModal && (
        <AddExpenseModal
          categories={CATEGORIES}
          staff={STAFF}
          onAdd={addExpense}
          onClose={() => setShowModal(false)}
        />
      )}
    </div>
  )
}

export default function Filters({
  search, onSearch,
  filterCategory, onCategory,
  filterStaff, onStaff,
  categories, staff,
}) {
  return (
    <div className={styles.bar}>
      <div className={styles.searchWrap}>
        <span className={styles.searchIcon}>🔍</span>
        <input
          className={styles.search}
          type="text"
          placeholder="Search expenses…"
          value={search}

        />
        {search && (

  )
}

import React, { useState } from 'react'
import styles from './ExpenseTable.module.css'

const fmt = (n) => new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(n)
const fmtDate = (d) => new Date(d + 'T00:00:00').toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })

function SortIcon({ field, sortField, sortDir }) {
  if (sortField !== field) return <span className={styles.sortNeutral}>⇅</span>
  return <span className={styles.sortActive}>{sortDir === 'asc' ? '↑' : '↓'}</span>
}

export default function ExpenseTable({ expenses, categories, filteredTotal, onDelete, sortField, sortDir, onSort }) {
  const [confirmId, setConfirmId] = useState(null)

  const catMap = Object.fromEntries(categories.map(c => [c.id, c]))

  const handleDelete = (id) => {
    if (confirmId === id) {
      onDelete(id)
      setConfirmId(null)
    } else {
      setConfirmId(id)
    }
  }

  return (
    <div className={styles.wrapper}>
      <div className={styles.tableHead}>
        <div className={styles.tableTitle}>
          Expenses
          <span className={styles.count}>{expenses.length} record{expenses.length !== 1 ? 's' : ''}</span>
        </div>
        <div className={styles.filteredTotal}>
          Showing: <strong>{fmt(filteredTotal)}</strong>
        </div>
      </div>

      {expenses.length === 0 ? (
        <div className={styles.empty}>
          <div className={styles.emptyIcon}>🔍</div>
          <div className={styles.emptyText}>No expenses match your filters</div>
          <div className={styles.emptySub}>Try adjusting your search or filters</div>
        </div>
      ) : (
        <div className={styles.tableScroll}>
          <table className={styles.table}>
            <thead>
              <tr>
                <th className={styles.th} onClick={() => onSort('date')}>
                  Date <SortIcon field="date" sortField={sortField} sortDir={sortDir} />
                </th>
                <th className={styles.th} onClick={() => onSort('staff')}>
                  Staff <SortIcon field="staff" sortField={sortField} sortDir={sortDir} />
                </th>
                <th className={styles.th} onClick={() => onSort('category')}>
                  Category <SortIcon field="category" sortField={sortField} sortDir={sortDir} />
                </th>
                <th className={styles.th} onClick={() => onSort('description')}>
                  Description <SortIcon field="description" sortField={sortField} sortDir={sortDir} />
                </th>
                <th className={`${styles.th} ${styles.right}`} onClick={() => onSort('amount')}>
                  Amount <SortIcon field="amount" sortField={sortField} sortDir={sortDir} />
                </th>
                <th className={styles.th}></th>
              </tr>
            </thead>
            <tbody>
              {expenses.map((exp, i) => {
                const cat = catMap[exp.category]
                return (
                  <tr key={exp.id} className={`${styles.row} ${i % 2 === 1 ? styles.rowAlt : ''}`}>
                    <td className={styles.td}>
                      <span className={styles.date}>{fmtDate(exp.date)}</span>
                    </td>
                    <td className={styles.td}>

                    </td>
                    <td className={`${styles.td} ${styles.actions}`}>
                      {confirmId === exp.id ? (
                        <div className={styles.confirmRow}>
                          <button className={styles.confirmYes} onClick={() => handleDelete(exp.id)}>Delete</button>
                          <button className={styles.confirmNo} onClick={() => setConfirmId(null)}>Cancel</button>
                        </div>
                      ) : (
                        <button className={styles.deleteBtn} onClick={() => handleDelete(exp.id)} title="Delete">🗑</button>
                      )}
                    </td>
                  </tr>
                )
              })}
            </tbody>
            <tfoot>
              <tr className={styles.footRow}>
                <td colSpan={4} className={`${styles.td} ${styles.footLabel}`}>

                </td>

                  {fmt(filteredTotal)}
                </td>
                <td className={styles.td}></td>
              </tr>
            </tfoot>
          </table>
        </div>
      )}
    </div>
  )
}

import React, { useState } from 'react'
import styles from './AddExpenseModal.module.css'

const today = () => new Date().toISOString().split('T')[0]

export default function AddExpenseModal({ categories, staff, onAdd, onClose }) {
  const [form, setForm] = useState({
    staff: staff[0],
    category: categories[0].id,
    description: '',
    amount: '',
    date: today(),
  })
  const [errors, setErrors] = useState({})

  const set = (k, v) => setForm(f => ({ ...f, [k]: v }))

  const validate = () => {
    const e = {}
    if (!form.description.trim()) e.description = 'Description is required'
    if (!form.amount || isNaN(+form.amount) || +form.amount <= 0) e.amount = 'Enter a valid amount'
    if (!form.date) e.date = 'Date is required'
    return e
  }

  const handleSubmit = (ev) => {
    ev.preventDefault()
    const e = validate()
    if (Object.keys(e).length) { setErrors(e); return }
    onAdd({ ...form, amount: parseFloat(form.amount) })
    onClose()
  }

  return (
    <div className={styles.overlay} onClick={e => e.target === e.currentTarget && onClose()}>
      <div className={styles.modal}>
        <div className={styles.modalHeader}>
          <div className={styles.modalTitle}>
            <span>➕</span> Add Expense
          </div>
          <button className={styles.closeBtn} onClick={onClose}>✕</button>
        </div>

        <form className={styles.form} onSubmit={handleSubmit} noValidate>
          <div className={styles.row2}>
            <div className={styles.field}>
              <label className={styles.label}>Staff Member</label>
              <select className={styles.input} value={form.staff} onChange={e => set('staff', e.target.value)}>
                {staff.map(s => <option key={s} value={s}>{s}</option>)}
              </select>
            </div>
            <div className={styles.field}>
              <label className={styles.label}>Date</label>
              <input
                className={`${styles.input} ${errors.date ? styles.inputErr : ''}`}
                type="date"
                value={form.date}
                max={today()}
                onChange={e => { set('date', e.target.value); setErrors(er => ({ ...er, date: '' })) }}
              />
              {errors.date && <span className={styles.err}>{errors.date}</span>}
            </div>
          </div>

          <div className={styles.field}>
            <label className={styles.label}>Category</label>
            <div className={styles.catGrid}>
              {categories.map(cat => (
                <button
                  key={cat.id}
                  type="button"
                  className={`${styles.catOption} ${form.category === cat.id ? styles.catSelected : ''}`}
                  style={{ '--c': cat.color, '--bg': cat.bg }}
                  onClick={() => set('category', cat.id)}
                >
                  <span>{cat.icon}</span>
                  <span>{cat.label}</span>
                </button>
              ))}
            </div>
          </div>

          <div className={styles.field}>
            <label className={styles.label}>Description</label>
            <input
              className={`${styles.input} ${errors.description ? styles.inputErr : ''}`}
              type="text"
              placeholder="What was this expense for?"
              value={form.description}
              onChange={e => { set('description', e.target.value); setErrors(er => ({ ...er, description: '' })) }}
            />
            {errors.description && <span className={styles.err}>{errors.description}</span>}
          </div>

          <div className={styles.field}>
            <label className={styles.label}>Amount (USD)</label>
            <div className={styles.amountWrap}>
              <span className={styles.dollar}>$</span>
              <input
                className={`${styles.input} ${styles.amountInput} ${errors.amount ? styles.inputErr : ''}`}
                type="number"
                min="0.01"
                step="0.01"
                placeholder="0.00"
                value={form.amount}
                onChange={e => { set('amount', e.target.value); setErrors(er => ({ ...er, amount: '' })) }}
              />
            </div>
            {errors.amount && <span className={styles.err}>{errors.amount}</span>}
          </div>

          <div className={styles.actions}>
            <button type="button" className={styles.cancelBtn} onClick={onClose}>Cancel</button>
            <button type="submit" className={styles.submitBtn}>Add Expense</button>
          </div>
        </form>
      </div>
    </div>
  )
}
