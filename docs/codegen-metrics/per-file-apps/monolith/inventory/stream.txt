import React, { useState, useMemo } from 'react'
import styles from './App.module.css'
import Sidebar from './components/Sidebar'
import Header from './components/Header'
import Dashboard from './components/Dashboard'
import ItemList from './components/ItemList'
import ItemDetail from './components/ItemDetail'
import AddItemModal from './components/AddItemModal'
import { initialItems } from './data/initialItems'

export default function App() {
  const [items, setItems] = useState(initialItems)
  const [search, setSearch] = useState('')
  const [filterCategory, setFilterCategory] = useState('All')
  const [filterLowStock, setFilterLowStock] = useState(false)
  const [selectedItem, setSelectedItem] = useState(null)
  const [showAddModal, setShowAddModal] = useState(false)
  const [view, setView] = useState('inventory') // 'dashboard' | 'inventory'

  const filtered = useMemo(() => {
    return items.filter(item => {
      const matchSearch = item.name.toLowerCase().includes(search.toLowerCase()) ||
        item.sku.toLowerCase().includes(search.toLowerCase())
      const matchCat = filterCategory === 'All' || item.category === filterCategory
      const matchLow = !filterLowStock || item.stock <= item.minStock
      return matchSearch && matchCat && matchLow
    })
  }, [items, search, filterCategory, filterLowStock])

  const handleUpdateStock = (id, delta, note) => {
    setItems(prev => prev.map(item => {
      if (item.id !== id) return item
      const newStock = Math.max(0, item.stock + delta)
      const today = new Date().toISOString().split('T')[0]
      return {
        ...item,
        stock: newStock,
        history: [...item.history, { date: today, change: delta, note: note || (delta > 0 ? 'Restock' : 'Adjustment') }]
      }
    }))
    setSelectedItem(prev => {
      if (!prev || prev.id !== id) return prev
      const updated = items.find(i => i.id === id)
      if (!updated) return prev
      const newStock = Math.max(0, updated.stock + delta)
      const today = new Date().toISOString().split('T')[0]
      return { ...updated, stock: newStock, history: [...updated.history, { date: today, change: delta, note: note || (delta > 0 ? 'Restock' : 'Adjustment') }] }
    })
  }

  const handleAddItem = (newItem) => {
    const id = Math.max(...items.map(i => i.id)) + 1
    setItems(prev => [...prev, { ...newItem, id, history: [{ date: new Date().toISOString().split('T')[0], change: newItem.stock, note: 'Initial stock' }] }])
    setShowAddModal(false)
  }

  const handleSelectItem = (item) => {
    const fresh = items.find(i => i.id === item.id) || item
    setSelectedItem(fresh)
  }

  // Keep selectedItem in sync with items
  const freshSelected = selectedItem ? items.find(i => i.id === selectedItem.id) || selectedItem : null

  return (
    <div className={styles.app}>
      <Sidebar view={view} setView={setView} items={items} />
      <div className={styles.main}>
        <Header
          search={search}
          setSearch={setSearch}
          filterCategory={filterCategory}
          setFilterCategory={setFilterCategory}
          filterLowStock={filterLowStock}
          setFilterLowStock={setFilterLowStock}
          onAdd={() => setShowAddModal(true)}
          view={view}
        />
        <div className={styles.content}>
          {view === 'dashboard' ? (
            <Dashboard items={items} onSelectItem={(item) => { handleSelectItem(item); setView('inventory') }} />
          ) : (
            <ItemList
              items={filtered}
              onSelect={handleSelectItem}
              selectedId={freshSelected?.id}
            />
          )}
        </div>
      </div>

      {freshSelected && (
        <ItemDetail
          item={freshSelected}
          onClose={() => setSelectedItem(null)}
          onUpdateStock={handleUpdateStock}
        />
      )}

      {showAddModal && (
        <AddItemModal
          onClose={() => setShowAddModal(false)}
          onAdd={handleAddItem}
        />
      )}
    </div>
  )
}

import React from 'react'
import styles from './Dashboard.module.css'
import { CATEGORIES } from '../data/initialItems'

export default function Dashboard({ items, onSelectItem }) {
  const totalUnits = items.reduce((s, i) => s + i.stock, 0)
  const totalValue = items.reduce((s, i) => s + i.stock * i.price, 0)
  const lowStock = items.filter(i => i.stock <= i.minStock)
  const outOfStock = items.filter(i => i.stock === 0)

  const byCategory = CATEGORIES.map(cat => {
    const catItems = items.filter(i => i.category === cat)
    return {
      name: cat,
      count: catItems.length,
      units: catItems.reduce((s, i) => s + i.stock, 0),
      value: catItems.reduce((s, i) => s + i.stock * i.price, 0),
    }
  }).filter(c => c.count > 0)

  const maxUnits = Math.max(...byCategory.map(c => c.units), 1)

  return (
    <div className={styles.dashboard}>
      <div className={styles.kpiRow}>
        <div className={styles.kpi}>
          <div className={styles.kpiIcon}>📦</div>
          <div>
            <div className={styles.kpiValue}>{items.length}</div>
            <div className={styles.kpiLabel}>Total SKUs</div>
          </div>
        </div>
        <div className={styles.kpi}>
          <div className={styles.kpiIcon}>🔢</div>
          <div>
            <div className={styles.kpiValue}>{totalUnits.toLocaleString()}</div>
            <div className={styles.kpiLabel}>Total Units</div>
          </div>
        </div>
        <div className={styles.kpi}>
          <div className={styles.kpiIcon}>💰</div>
          <div>
            <div className={styles.kpiValue}>${totalValue.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</div>
            <div className={styles.kpiLabel}>Stock Value</div>
          </div>
        </div>
        <div className={`${styles.kpi} ${lowStock.length > 0 ? styles.warn : ''}`}>
          <div className={styles.kpiIcon}>⚠️</div>
          <div>
            <div className={styles.kpiValue}>{lowStock.length}</div>
            <div className={styles.kpiLabel}>Low Stock Items</div>
          </div>
        </div>
        <div className={`${styles.kpi} ${outOfStock.length > 0 ? styles.danger : ''}`}>
          <div className={styles.kpiIcon}>🚫</div>
          <div>
            <div className={styles.kpiValue}>{outOfStock.length}</div>
            <div className={styles.kpiLabel}>Out of Stock</div>
          </div>
        </div>
      </div>

      <div className={styles.row2}>
        <div className={styles.card}>
          <h2 className={styles.cardTitle}>Stock by Category</h2>
          <div className={styles.barChart}>
            {byCategory.map(cat => (
              <div key={cat.name} className={styles.barRow}>
                <div className={styles.barLabel}>{cat.name}</div>
                <div className={styles.barTrack}>
                  <div
                    className={styles.barFill}
                    style={{ width: `${(cat.units / maxUnits) * 100}%` }}
                  />
                </div>
                <div className={styles.barValue}>{cat.units}</div>
              </div>
            ))}
          </div>
        </div>

        <div className={styles.card}>
          <h2 className={styles.cardTitle}>⚠ Low Stock Alerts</h2>
          {lowStock.length === 0 ? (
            <div className={styles.empty}>All items are well stocked ✓</div>
          ) : (
            <div className={styles.alertList}>
              {lowStock.map(item => (
                <button key={item.id} className={styles.alertItem} onClick={() => onSelectItem(item)}>
                  <div className={styles.alertLeft}>
                    <div className={styles.alertName}>{item.name}</div>
                    <div className={styles.alertSku}>{item.sku} · {item.category}</div>

import React, { useState } from 'react'
import styles from './ItemDetail.module.css'

function stockStatus(item) {
  if (item.stock === 0) return 'out'
  if (item.stock <= item.minStock) return 'low'
  return 'ok'
}

const statusLabel = { ok: 'In Stock', low: 'Low Stock', out: 'Out of Stock' }
const statusColor = { ok: 'green', low: 'yellow', out: 'red' }

export default function ItemDetail({ item, onClose, onUpdateStock }) {
  const [delta, setDelta] = useState('')
  const [note, setNote] = useState('')
  const [mode, setMode] = useState('add') // 'add' | 'remove'

  const status = stockStatus(item)

  const handleApply = () => {
    const num = parseInt(delta)
    if (!num || num <= 0) return
    const change = mode === 'add' ? num : -num
    onUpdateStock(item.id, change, note)
    setDelta('')
    setNote('')
  }

  return (
    <div className={styles.overlay}>
      <div className={styles.panel}>
        <div className={styles.panelHeader}>
          <div>
            <h2 className={styles.panelTitle}>{item.name}</h2>

            <div className={styles.stockSub}>units in stock</div>

            </div>

              <div className={styles.infoLabel}>Location</div>
              <div className={styles.infoValue}>{item.location}</div>
            </div>

          </div>

          {item.description && (
            <div className={styles.desc}>{item.description}</div>
          )}

            <h3 className={styles.sectionTitle}>Adjust Stock</h3>

              >+ Add</button>
              <button
                className={`${styles.modeBtn} ${mode === 'remove' ? styles.modeActiveRed : ''}`}
                onClick={() => setMode('remove')}

                className={styles.deltaInput}
                type="number"

                </div>

            </div>
          </div>
        </div>
      </div>
    </div>
  )
}

import React, { useState } from 'react'
import styles from './AddItemModal.module.css'
import { CATEGORIES } from '../data/initialItems'

const empty = { name: '', sku: '', category: CATEGORIES[0], stock: '', minStock: '', price: '', location: '', description: '' }

export default function AddItemModal({ onClose, onAdd }) {
  const [form, setForm] = useState(empty)
  const [errors, setErrors] = useState({})

  const set = (k, v) => setForm(f => ({ ...f, [k]: v }))

  const validate = () => {
    const e = {}
    if (!form.name.trim()) e.name = 'Required'
    if (!form.sku.trim()) e.sku = 'Required'
    if (form.stock === '' || isNaN(form.stock) || Number(form.stock) < 0) e.stock = 'Valid number required'
    if (form.minStock === '' || isNaN(form.minStock) || Number(form.minStock) < 0) e.minStock = 'Valid number required'
    if (form.price === '' || isNaN(form.price) || Number(form.price) < 0) e.price = 'Valid price required'
    return e
  }

  const handleSubmit = (e) => {
    e.preventDefault()
    const errs = validate()
    if (Object.keys(errs).length) { setErrors(errs); return }
    onAdd({
      name: form.name.trim(),
      sku: form.sku.trim().toUpperCase(),
      category: form.category,
      stock: parseInt(form.stock),
      minStock: parseInt(form.minStock),

      <div className={styles.modal}>

              {errors.sku && <span className={styles.errMsg}>{errors.sku}</span>}
            </div>
          </div>

          <div className={styles.row}>
            <div className={styles.field}>
              <label className={styles.label}>Category</label>

            </div>

import styles from './StockBadge.module.css';

export default function StockBadge({ quantity, minStock }) {
  const ratio = minStock > 0 ? quantity / minStock : 1;

  let level, label;
  if (quantity === 0) {
    level = 'out';
    label = 'Out of Stock';
  } else if (ratio <= 1) {
    level = 'low';
    label = 'Low Stock';
  } else if (ratio <= 2) {
    level = 'medium';
    label = 'Medium';
  } else {
    level = 'good';
    label = 'In Stock';
  }

  return (
    <span className={`${styles.badge} ${styles[level]}`}>
      <span className={styles.dot} />
      {label}
    </span>
  );
}

import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import initialItems from '../data/initialItems';

const useInventoryStore = create(
  persist(
    (set, get) => ({
      items: initialItems,
      searchQuery: '',
      filterCategory: 'All',
      filterLowStock: false,
      selectedItemId: null,
      view: 'dashboard', // 'dashboard' | 'inventory' | 'detail'

      setSearchQuery: (q) => set({ searchQuery: q }),
      setFilterCategory: (c) => set({ filterCategory: c }),
      setFilterLowStock: (v) => set({ filterLowStock: v }),
      setSelectedItemId: (id) => set({ selectedItemId: id }),
      setView: (v) => set({ view: v }),

      openDetail: (id) => set({ selectedItemId: id, view: 'detail' }),
      goBack: () => set({ view: 'inventory', selectedItemId: null }),

      addItem: (item) =>
        set((state) => ({ items: [item, ...state.items] })),

      updateQuantity: (id, delta, reason) =>
        set((state) => ({
          items: state.items.map((item) => {
            if (item.id !== id) return item;
            const newQty = Math.max(0, item.quantity + delta);
            return {
              ...item,
              quantity: newQty,
              lastUpdated: new Date().toISOString().split('T')[0],
              history: [
                {
                  date: new Date().toISOString().split('T')[0],
                  change: delta,
                  reason: reason || (delta > 0 ? 'Restock' : 'Adjustment'),
                  user: 'Staff',
                },
                ...(item.history || []),
              ],
            };
          }),
        })),

      deleteItem: (id) =>
        set((state) => ({
          items: state.items.filter((item) => item.id !== id),
          selectedItemId: null,
          view: 'inventory',
        })),

      getFilteredItems: () => {
        const { items, searchQuery, filterCategory, filterLowStock } = get();
        return items.filter((item) => {
          const q = searchQuery.toLowerCase();
          const matchSearch =
            !q ||
            item.name.toLowerCase().includes(q) ||
            item.sku.toLowerCase().includes(q) ||
            item.category.toLowerCase().includes(q) ||
            item.location.toLowerCase().includes(q);
          const matchCategory =
            filterCategory === 'All' || item.category === filterCategory;
          const matchLow = !filterLowStock || item.quantity <= item.minStock;
          return matchSearch && matchCategory && matchLow;
        });
      },

      getStats: () => {
        const { items } = get();
        const totalItems = items.length;
        const totalUnits = items.reduce((s, i) => s + i.quantity, 0);
        const totalValue = items.reduce((s, i) => s + i.quantity * i.price, 0);
        const lowStock = items.filter((i) => i.quantity <= i.minStock && i.quantity > 0).length;
        const outOfStock = items.filter((i) => i.quantity === 0).length;
        return { totalItems, totalUnits, totalValue, lowStock, outOfStock };
      },

      getCategories: () => {
        const { items } = get();
        const cats = [...new Set(items.map((i) => i.category))].sort();
        return ['All', ...cats];
      },
    }),
    {
      name: 'inventory-store',
      partialize: (state) => ({ items: state.items }),
    }
  )
);

export default useInventoryStore;
