<<<FILE path="src/store/useStore.js">>>
import { create } from "zustand";
import { seedJobs } from "../data/jobs";

let nextId = 200;
const genId = () => `JOB-${++nextId}`;

const useStore = create((set, get) => ({
  jobs: seedJobs(),

  addJob: (jobData) => {
    const job = {
      ...jobData,
      id: genId(),
      created: new Date().toISOString(),
      notes: "",
    };
    set((s) => ({ jobs: [job, ...s.jobs] }));
    return job.id;
  },

  updateJob: (id, patch) => {
    set((s) => ({
      jobs: s.jobs.map((j) => (j.id === id ? { ...j, ...patch } : j)),
    }));
  },

  deleteJob: (id) => {
    set((s) => ({ jobs: s.jobs.filter((j) => j.id !== id) }));
  },

  getJob: (id) => get().jobs.find((j) => j.id === id),
}));

export default useStore;

import React, { useState, useMemo } from "react";
import { isPast, parseISO } from "date-fns";
import useStore from "../store/useStore";
import StatsBar from "../components/StatsBar";
import JobTable from "../components/JobTable";
import CreateJobModal from "../components/CreateJobModal";
import SearchBar from "../components/SearchBar";
import FilterBar from "../components/FilterBar";
import styles from "./Dashboard.module.css";
import { Plus } from "lucide-react";

export default function Dashboard({ navigate }) {
  const jobs = useStore((s) => s.jobs);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("All");
  const [priorityFilter, setPriorityFilter] = useState("All");
  const [categoryFilter, setCategoryFilter] = useState("All");
  const [showCreate, setShowCreate] = useState(false);

  const overdue = useMemo(
    () =>
      jobs.filter(
        (j) =>
          j.status !== "Completed" &&
          j.status !== "Cancelled" &&
          isPast(parseISO(j.due))
      ),
    [jobs]
  );

  const filtered = useMemo(() => {
    const q = search.toLowerCase();
    return jobs.filter((j) => {
      const matchSearch =
        !q ||
        j.title.toLowerCase().includes(q) ||
        j.id.toLowerCase().includes(q) ||
        j.assignee.toLowerCase().includes(q);
      const matchStatus = statusFilter === "All" || j.status === statusFilter;
      const matchPriority = priorityFilter === "All" || j.priority === priorityFilter;
      const matchCategory = categoryFilter === "All" || j.category === categoryFilter;
      return matchSearch && matchStatus && matchPriority && matchCategory;
    });
  }, [jobs, search, statusFilter, priorityFilter, categoryFilter]);

  return (
    <div className={styles.page}>
      <header className={styles.header}>
        <div className={styles.headerLeft}>
          <span className={styles.logo}>⚙️</span>
          <div>
            <h1 className={styles.title}>Operations Dashboard</h1>
            <p className={styles.subtitle}>Supervisor View · {jobs.length} total jobs</p>
          </div>
        </div>
        <button className={styles.createBtn} onClick={() => setShowCreate(true)}>
          <Plus size={16} />
          New Job
        </button>
      </header>

      <StatsBar jobs={jobs} overdue={overdue} />

      <div className={styles.controls}>
        <SearchBar value={search} onChange={setSearch} />
        <FilterBar
          statusFilter={statusFilter} setStatusFilter={setStatusFilter}
          priorityFilter={priorityFilter} setPriorityFilter={setPriorityFilter}
          categoryFilter={categoryFilter} setCategoryFilter={setCategoryFilter}
        />
      </div>

      <div className={styles.tableWrap}>
        <JobTable jobs={filtered} navigate={navigate} />
      </div>

      {showCreate && (
        <CreateJobModal onClose={() => setShowCreate(false)} navigate={navigate} />
      )}
    </div>
  );
}

import React, { useState } from "react";
import { format, parseISO, isPast } from "date-fns";
import useStore from "../store/useStore";
import StatusBadge from "../components/StatusBadge";
import PriorityBadge from "../components/PriorityBadge";
import styles from "./JobDetail.module.css";
import { ArrowLeft, Trash2, Save, AlertTriangle } from "lucide-react";
import { STATUSES, PRIORITIES, ASSIGNEES } from "../data/jobs";

export default function JobDetail({ jobId, navigate }) {
  const job = useStore((s) => s.getJob(jobId));
  const updateJob = useStore((s) => s.updateJob);
  const deleteJob = useStore((s) => s.deleteJob);

  const [form, setForm] = useState(job ? { ...job } : null);
  const [saved, setSaved] = useState(false);
  const [confirmDelete, setConfirmDelete] = useState(false);

  if (!job || !form) {
    return (
      <div className={styles.notFound}>
        <p>Job not found.</p>
        <button onClick={() => navigate("dashboard")}>← Back</button>
      </div>
    );
  }

  const isOverdue =
    form.status !== "Completed" &&
    form.status !== "Cancelled" &&
    isPast(parseISO(form.due));

  const set = (key, val) => {
    setForm((f) => ({ ...f, [key]: val }));
    setSaved(false);
  };

  const handleSave = () => {
    updateJob(jobId, form);
    setSaved(true);
    setTimeout(() => setSaved(false), 2000);
  };

  const handleDelete = () => {
    deleteJob(jobId);
    navigate("dashboard");
  };

  return (
    <div className={styles.page}>
      <div className={styles.topBar}>
        <button className={styles.back} onClick={() => navigate("dashboard")}>
          <ArrowLeft size={16} /> Back to Dashboard
        </button>
        <div className={styles.topActions}>
          {saved && <span className={styles.savedMsg}>✓ Saved</span>}
          <button className={styles.saveBtn} onClick={handleSave}>
            <Save size={15} /> Save Changes
          </button>
          <button className={styles.deleteBtn} onClick={() => setConfirmDelete(true)}>
            <Trash2 size={15} />
          </button>
        </div>
      </div>

      {isOverdue && (
        <div className={styles.overdueBanner}>
          <AlertTriangle size={16} />
          This job is <strong>overdue</strong>. Due date was{" "}
          {format(parseISO(form.due), "MMM d, yyyy")}.
        </div>
      )}

      <div className={styles.card}>
        <div className={styles.cardHeader}>
          <div>
            <span className={styles.jobId}>{form.id}</span>
            <input
              className={styles.titleInput}
              value={form.title}
              onChange={(e) => set("title", e.target.value)}
            />
          </div>
          <div className={styles.badges}>
            <StatusBadge status={form.status} />
            <PriorityBadge priority={form.priority} />
          </div>
        </div>

        <div className={styles.grid}>
          <Field label="Status">
            <select value={form.status} onChange={(e) => set("status", e.target.value)}>
              {STATUSES.map((s) => <option key={s}>{s}</option>)}
            </select>
          </Field>
          <Field label="Priority">
            <select value={form.priority} onChange={(e) => set("priority", e.target.value)}>
              {PRIORITIES.map((p) => <option key={p}>{p}</option>)}
            </select>
          </Field>
          <Field label="Assignee">
            <select value={form.assignee} onChange={(e) => set("assignee", e.target.value)}>
              {ASSIGNEES.map((a) => <option key={a}>{a}</option>)}
            </select>
          </Field>
          <Field label="Category">
            <input value={form.category} onChange={(e) => set("category", e.target.value)} />
          </Field>
          <Field label="Due Date">
            <input
              type="date"
              value={form.due.slice(0, 10)}
              onChange={(e) => set("due", new Date(e.target.value).toISOString())}
            />
          </Field>
          <Field label="Created">
            <input value={format(parseISO(form.created), "MMM d, yyyy HH:mm")} readOnly />
          </Field>
        </div>

        <Field label="Description">
          <textarea
            rows={4}
            value={form.description}
            onChange={(e) => set("description", e.target.value)}
          />
        </Field>

        <Field label="Notes / Updates">
          <textarea
            rows={4}
            value={form.notes}
            onChange={(e) => set("notes", e.target.value)}
            placeholder="Add internal notes or status updates…"
          />
        </Field>
      </div>

      {confirmDelete && (
        <div className={styles.overlay}>
          <div className={styles.dialog}>
            <h3>Delete Job?</h3>
            <p>This will permanently remove <strong>{form.id}</strong>. This cannot be undone.</p>
            <div className={styles.dialogActions}>
              <button className={styles.cancelBtn} onClick={() => setConfirmDelete(false)}>Cancel</button>
              <button className={styles.confirmDeleteBtn} onClick={handleDelete}>Delete</button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

function Field({ label, children }) {
  return (
    <div className={styles.field}>
      <label className={styles.label}>{label}</label>
      {children}
    </div>
  );
}

const COLS = [
  { key: "id", label: "Job ID" },
  { key: "title", label: "Title" },
  { key: "category", label: "Category" },
  { key: "status", label: "Status" },
  { key: "priority", label: "Priority" },
  { key: "assignee", label: "Assignee" },
  { key: "due", label: "Due Date" },
];

export default function JobTable({ jobs, navigate }) {
  const [sort, setSort] = useState({ key: "due", dir: "asc" });

  const toggleSort = (key) => {
    setSort((s) => ({ key, dir: s.key === key && s.dir === "asc" ? "desc" : "asc" }));
  };

  const sorted = [...jobs].sort((a, b) => {
    let av = a[sort.key] ?? "";
    let bv = b[sort.key] ?? "";
    if (sort.key === "due" || sort.key === "created") {
      av = new Date(av);
      bv = new Date(bv);
    } else {
      av = av.toString().toLowerCase();
      bv = bv.toString().toLowerCase();
    }
    if (av < bv) return sort.dir === "asc" ? -1 : 1;
    if (av > bv) return sort.dir === "asc" ? 1 : -1;
    return 0;
  });

  return (
    <div className={styles.tableContainer}>
      <table className={styles.table}>
        <thead>
          <tr>
            {COLS.map((c) => (
              <th key={c.key} onClick={() => toggleSort(c.key)} className={styles.th}>
                <span className={styles.thInner}>
                  {c.label}
                  <span className={styles.sortIcon}>
                    {sort.key === c.key
                      ? sort.dir === "asc" ? <ChevronUp size={13} /> : <ChevronDown size={13} />
                      : <ChevronDown size={13} className={styles.sortFade} />}
                  </span>
                </span>
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {sorted.length === 0 && (
            <tr>
              <td colSpan={COLS.length} className={styles.empty}>
                No jobs match your filters.
              </td>
            </tr>
          )}
          {sorted.map((job) => {
            const overdue =
              job.status !== "Completed" &&
              job.status !== "Cancelled" &&
              isPast(parseISO(job.due));
            return (
              <tr
                key={job.id}
                className={`${styles.row} ${overdue ? styles.overdueRow : ""}`}
                onClick={() => navigate("detail", { jobId: job.id })}
              >
                <td className={styles.td}>
                  <span className={styles.jobId}>{job.id}</span>
                </td>
                <td className={styles.td}>
                  <span className={styles.jobTitle}>{job.title}</span>
                  {overdue && (
                    <span className={styles.overdueTag}>
                      <AlertTriangle size={11} /> Overdue
                    </span>
                  )}
                </td>
                <td className={styles.td}>
                  <span className={styles.category}>{job.category}</span>
                </td>
                <td className={styles.td}>
                  <StatusBadge status={job.status} />
                </td>
                <td className={styles.td}>
                  <PriorityBadge priority={job.priority} />
                </td>
                <td className={styles.td}>
                  <span className={styles.assignee}>{job.assignee}</span>
                </td>
                <td className={styles.td}>
                  <span className={overdue ? styles.dueDateOverdue : styles.dueDate}>
                    {format(parseISO(job.due), "MMM d, yyyy")}
                  </span>
                </td>
              </tr>
            );
          })}

  updateJob: (id, patch) => {
    set((s) => ({
      jobs: s.jobs.map((j) => (j.id === id ? { ...j, ...patch } : j)),
    }));
  },

  deleteJob: (id) => {
    set((s) => ({ jobs: s.jobs.filter((j) => j.id !== id) }));
  },

  getJob: (id) => get().jobs.find((j) => j.id === id),
}));

export default useStore;
