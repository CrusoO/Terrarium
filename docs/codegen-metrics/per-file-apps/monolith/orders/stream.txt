export const STATUSES = ['Pending', 'Paid', 'Refunded', 'Cancelled']

export const ITEMS_CATALOG = [
  { id: 'p1', name: 'Espresso', price: 3.5 },
  { id: 'p2', name: 'Cappuccino', price: 4.5 },
  { id: 'p3', name: 'Latte', price: 5.0 },
  { id: 'p4', name: 'Croissant', price: 3.0 },
  { id: 'p5', name: 'Bagel', price: 3.5 },
  { id: 'p6', name: 'Muffin', price: 2.5 },
  { id: 'p7', name: 'Sandwich', price: 7.5 },
  { id: 'p8', name: 'Salad Bowl', price: 9.0 },
  { id: 'p9', name: 'Orange Juice', price: 4.0 },
  { id: 'p10', name: 'Green Tea', price: 3.0 },
]

function randomItem(arr) { return arr[Math.floor(Math.random() * arr.length)] }
function randomInt(min, max) { return Math.floor(Math.random() * (max - min + 1)) + min }

let idCounter = 1000

export function generateOrders(count = 40) {
  const names = ['Alice Johnson','Bob Martinez','Carol White','David Lee','Eva Brown',
    'Frank Wilson','Grace Kim','Henry Davis','Iris Chen','Jack Taylor']
  const now = Date.now()
  const DAY = 86400000
  const orders = []
  for (let i = 0; i < count; i++) {
    const daysAgo = randomInt(0, 6)
    const ts = now - daysAgo * DAY - randomInt(0, DAY - 1)
    const itemCount = randomInt(1, 4)
    const items = []
    for (let j = 0; j < itemCount; j++) {
      const cat = randomItem(ITEMS_CATALOG)
      const existing = items.find(x => x.id === cat.id)
      if (existing) existing.qty += 1
      else items.push({ ...cat, qty: randomInt(1, 3) })
    }
    const total = items.reduce((s, x) => s + x.price * x.qty, 0)
    orders.push({
      id: `ORD-${++idCounter}`,
      customer: randomItem(names),
      status: randomItem(STATUSES),
      items,
      total: +total.toFixed(2),
      createdAt: ts,
      note: Math.random() > 0.7 ? 'Please pack separately.' : '',
    })
  }
  return orders.sort((a, b) => b.createdAt - a.createdAt)
}

export function nextId() { return `ORD-${++idCounter}` }

import { useState, useCallback } from 'react'
import { generateOrders, nextId } from '../data/seed'

const initialOrders = generateOrders(40)

let _orders = initialOrders
let _listeners = []

function notify() { _listeners.forEach(fn => fn([..._orders])) }

export function useOrders() {
  const [orders, setOrders] = useState([..._orders])

  const subscribe = useCallback(() => {
    const fn = (next) => setOrders(next)
    _listeners.push(fn)
    return () => { _listeners = _listeners.filter(l => l !== fn) }
  }, [])

  useState(() => { return subscribe() })

  const addOrder = useCallback((order) => {
    const newOrder = { ...order, id: nextId(), createdAt: Date.now() }
    _orders = [newOrder, ..._orders]
    notify()
  }, [])

  const updateStatus = useCallback((id, status) => {
    _orders = _orders.map(o => o.id === id ? { ...o, status } : o)
    notify()
  }, [])

  return { orders, addOrder, updateStatus }
}

export function formatCurrency(n) {
  return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(n)
}

export function formatDate(ts) {
  return new Intl.DateTimeFormat('en-US', {
    month: 'short', day: 'numeric', year: 'numeric',
    hour: '2-digit', minute: '2-digit',
  }).format(new Date(ts))
}

export function formatDay(ts) {
  const d = new Date(ts)
  const today = new Date()
  const yesterday = new Date(today); yesterday.setDate(today.getDate() - 1)
  if (d.toDateString() === today.toDateString()) return 'Today'
  if (d.toDateString() === yesterday.toDateString()) return 'Yesterday'
  return new Intl.DateTimeFormat('en-US', { weekday: 'short', month: 'short', day: 'numeric' }).format(d)
}

export function dayKey(ts) {
  const d = new Date(ts)
  return `${d.getFullYear()}-${d.getMonth()}-${d.getDate()}`
}

import React, { useState } from 'react'
import Header from './components/Header'
import OrderList from './components/OrderList'
import OrderDetail from './components/OrderDetail'
import CreateOrder from './components/CreateOrder'
import DailyTotals from './components/DailyTotals'
import { useOrders } from './store/useStore'
import styles from './App.module.css'

export default function App() {
  const { orders, addOrder, updateStatus } = useOrders()
  const [search, setSearch] = useState('')
  const [statusFilter, setStatusFilter] = useState('All')
  const [selectedId, setSelectedId] = useState(null)
  const [showCreate, setShowCreate] = useState(false)
  const [activeTab, setActiveTab] = useState('orders') // 'orders' | 'totals'

  const filtered = orders.filter(o => {
    const matchSearch = search === '' ||
      o.id.toLowerCase().includes(search.toLowerCase()) ||
      o.customer.toLowerCase().includes(search.toLowerCase())
    const matchStatus = statusFilter === 'All' || o.status === statusFilter
    return matchSearch && matchStatus
  })

  const selectedOrder = orders.find(o => o.id === selectedId) || null

  return (
    <div className={styles.app}>
      <Header
        search={search}
        onSearch={setSearch}
        statusFilter={statusFilter}
        onStatusFilter={setStatusFilter}
        onCreateOrder={() => setShowCreate(true)}
        activeTab={activeTab}
        onTabChange={setActiveTab}
      />
      <main className={styles.main}>
        {activeTab === 'orders' ? (
          <div className={styles.layout}>
            <OrderList
              orders={filtered}
              selectedId={selectedId}
              onSelect={setSelectedId}
            />
            {selectedOrder ? (
              <OrderDetail
                order={selectedOrder}
                onClose={() => setSelectedId(null)}
                onStatusChange={updateStatus}
              />
            ) : (
              <div className={styles.empty}>
                <div className={styles.emptyInner}>
                  <span className={styles.emptyIcon}>📋</span>
                  <p>Select an order to view details</p>
                </div>
              </div>
            )}
          </div>
        ) : (
          <DailyTotals orders={orders} />
        )}
      </main>
      {showCreate && (
        <CreateOrder
          onClose={() => setShowCreate(false)}
          onSubmit={(order) => { addOrder(order); setShowCreate(false) }}
        />
      )}
    </div>
  )
}

import React, { useState } from 'react'
import { ITEMS_CATALOG } from '../data/seed'
import { formatCurrency } from '../utils/format'
import styles from './CreateOrder.module.css'

export default function CreateOrder({ onClose, onSubmit }) {
  const [customer, setCustomer] = useState('')
  const [note, setNote] = useState('')
  const [cart, setCart] = useState([])
  const [errors, setErrors] = useState({})

  const addItem = (item) => {
    setCart(prev => {
      const ex = prev.find(x => x.id === item.id)
      if (ex) return prev.map(x => x.id === item.id ? { ...x, qty: x.qty + 1 } : x)
      return [...prev, { ...item, qty: 1 }]
    })
  }

  const removeItem = (id) => setCart(prev => prev.filter(x => x.id !== id))

  const updateQty = (id, qty) => {
    if (qty < 1) { removeItem(id); return }
    setCart(prev => prev.map(x => x.id === id ? { ...x, qty } : x))
  }

  const subtotal = cart.reduce((s, i) => s + i.price * i.qty, 0)
  const tax = subtotal * 0.08
  const total = subtotal + tax

  const validate = () => {
    const e = {}
    if (!customer.trim()) e.customer = 'Customer name is required'
    if (cart.length === 0) e.cart = 'Add at least one item'
    setErrors(e)
    return Object.keys(e).length === 0
  }

  const handleSubmit = (e) => {
    e.preventDefault()
    if (!validate()) return
    onSubmit({
      customer: customer.trim(),
      items: cart,
      total: +total.toFixed(2),
      status: 'Pending',
      note: note.trim(),
    })
  }

  return (
    <div className={styles.overlay} onClick={e => e.target === e.currentTarget && onClose()}>
      <div className={styles.modal}>
        <div className={styles.modalHeader}>
          <h2 className={styles.modalTitle}>Create New Order</h2>
          <button className={styles.closeBtn} onClick={onClose}>✕</button>
        </div>

        <form onSubmit={handleSubmit} className={styles.form}>
          <div className={styles.body}>
            <div className={styles.left}>
              <div className={styles.field}>
                <label className={styles.label}>Customer Name *</label>
                <input
                  className={`${styles.input} ${errors.customer ? styles.inputError : ''}`}
                  type="text"
                  placeholder="Enter customer name"
                  value={customer}
                  onChange={e => { setCustomer(e.target.value); setErrors(p => ({ ...p, customer: '' })) }}
                />
                {errors.customer && <span className={styles.error}>{errors.customer}</span>}
              </div>

              <div className={styles.field}>
                <label className={styles.label}>Note (optional)</label>
                <textarea
                  className={styles.textarea}
                  placeholder="Any special instructions…"
                  value={note}
                  onChange={e => setNote(e.target.value)}
                  rows={2}
                />
              </div>

              <div className={styles.field}>
                <label className={styles.label}>Menu Items</label>
                {errors.cart && <span className={styles.error}>{errors.cart}</span>}
                <div className={styles.catalog}>
                  {ITEMS_CATALOG.map(item => (
                    <button
                      key={item.id}
                      type="button"
                      className={styles.catalogItem}
                      onClick={() => addItem(item)}
                    >
                      <span className={styles.catalogName}>{item.name}</span>
                      <span className={styles.catalogPrice}>{formatCurrency(item.price)}</span>
                      <span className={styles.addIcon}>+</span>
                    </button>
                  ))}
                </div>
              </div>
            </div>

            <div className={styles.right}>
              <div className={styles.cartHeader}>
                <span className={styles.label}>Cart</span>
                <span className={styles.cartCount}>{cart.length} item{cart.length !== 1 ? 's' : ''}</span>
              </div>

              {cart.length === 0 ? (
                <div className={styles.cartEmpty}>
                  <span>🛒</span>
                  <p>No items added yet</p>
                </div>
              ) : (
                <div className={styles.cartItems}>
                  {cart.map(item => (
                    <div key={item.id} className={styles.cartItem}>
                      <div className={styles.cartItemInfo}>
                        <span className={styles.cartItemName}>{item.name}</span>
                        <span className={styles.cartItemPrice}>{formatCurrency(item.price)}</span>
                      </div>
                      <div className={styles.cartItemControls}>
                        <button type="button" className={styles.qtyBtn} onClick={() => updateQty(item.id, item.qty - 1)}>−</button>
                        <span className={styles.qty}>{item.qty}</span>
                        <button type="button" className={styles.qtyBtn} onClick={() => updateQty(item.id, item.qty + 1)}>+</button>
                        <span className={styles.cartItemTotal}>{formatCurrency(item.price * item.qty)}</span>
                        <button type="button" className={styles.removeBtn} onClick={() => removeItem(item.id)}>✕</button>
                      </div>
                    </div>
                  ))}
                </div>
              )}

              <div className={styles.cartTotals}>
                <div className={styles.totalRow}>
                  <span>Subtotal</span>
                  <span>{formatCurrency(subtotal)}</span>
                </div>
                <div className={styles.totalRow}>
                  <span>Tax (8%)</span>
                  <span>{formatCurrency(tax)}</span>
                </div>
                <div className={`${styles.totalRow} ${styles.grandTotal}`}>
                  <span>Total</span>
                  <span>{formatCurrency(total)}</span>
                </div>
              </div>
            </div>
          </div>

          <div className={styles.footer}>
            <button type="button" className={styles.cancelBtn} onClick={onClose}>Cancel</button>
            <button type="submit" className={styles.submitBtn}>Create Order</button>
          </div>
        </form>
      </div>
    </div>
  )
}

import React from 'react'
import { formatCurrency, formatDay, dayKey } from '../utils/format'
import styles from './DailyTotals.module.css'

const STATUS_COLORS = {
  Pending: { bg: 'var(--warning-light)', color: 'var(--warning)' },
  Paid: { bg: 'var(--success-light)', color: 'var(--success)' },
  Refunded: { bg: '#e0e7ff', color: '#4338ca' },
  Cancelled: { bg: 'var(--danger-light)', color: 'var(--danger)' },
}

export default function DailyTotals({ orders }) {
  const grouped = {}
  orders.forEach(o => {
    const key = dayKey(o.createdAt)
    if (!grouped[key]) grouped[key] = { ts: o.createdAt, orders: [] }
    grouped[key].orders.push(o)
  })

  const days = Object.entries(grouped)
    .sort((a, b) => b[1].ts - a[1].ts)
    .map(([key, { ts, orders: dayOrders }]) => {
      const revenue = dayOrders.reduce((s, o) => s + o.total, 0)
      const paid = dayOrders.filter(o => o.status === 'Paid').reduce((s, o) => s + o.total, 0)
      const pending = dayOrders.filter(o => o.status === 'Pending').reduce((s, o) => s + o.total, 0)
      const refunded = dayOrders.filter(o => o.status === 'Refunded').reduce((s, o) => s + o.total, 0)
      const cancelled = dayOrders.filter(o => o.status === 'Cancelled').reduce((s, o) => s + o.total, 0)
      const statusCounts = {}
      dayOrders.forEach(o => { statusCounts[o.status] = (statusCounts[o.status] || 0) + 1 })
      return { key, ts, dayOrders, revenue, paid, pending, refunded, cancelled, statusCounts }
    })

  const grandTotal = days.reduce((s, d) => s + d.revenue, 0)
  const grandPaid = days.reduce((s, d) => s + d.paid, 0)
  const grandOrders = orders.length

  return (
    <div className={styles.container}>
      <div className={styles.summaryCards}>
        <div className={styles.summaryCard}>
          <div className={styles.summaryLabel}>Total Revenue</div>
          <div className={styles.summaryValue}>{formatCurrency(grandTotal)}</div>
          <div className={styles.summaryNote}>{grandOrders} orders total</div>
        </div>
        <div className={`${styles.summaryCard} ${styles.summaryGreen}`}>
          <div className={styles.summaryLabel}>Paid Revenue</div>
          <div className={styles.summaryValue}>{formatCurrency(grandPaid)}</div>
          <div className={styles.summaryNote}>{orders.filter(o => o.status === 'Paid').length} paid orders</div>
        </div>
        <div className={`${styles.summaryCard} ${styles.summaryYellow}`}>
          <div className={styles.summaryLabel}>Pending</div>
          <div className={styles.summaryValue}>{formatCurrency(days.reduce((s, d) => s + d.pending, 0))}</div>
          <div className={styles.summaryNote}>{orders.filter(o => o.status === 'Pending').length} pending orders</div>
        </div>
        <div className={`${styles.summaryCard} ${styles.summaryRed}`}>
          <div className={styles.summaryLabel}>Refunded</div>
          <div className={styles.summaryValue}>{formatCurrency(days.reduce((s, d) => s + d.refunded, 0))}</div>
          <div className={styles.summaryNote}>{orders.filter(o => o.status === 'Refunded').length} refunded</div>
        </div>
      </div>

      <div className={styles.days}>
        {days.map(day => (
          <div key={day.key} className={styles.dayCard}>
            <div className={styles.dayHeader}>
              <div>
                <div className={styles.dayLabel}>{formatDay(day.ts)}</div>
                <div className={styles.dayMeta}>{day.dayOrders.length} order{day.dayOrders.length !== 1 ? 's' : ''}</div>
              </div>
              <div className={styles.dayRevenue}>{formatCurrency(day.revenue)}</div>
            </div>

            <div className={styles.dayStats}>
              {Object.entries(day.statusCounts).map(([status, count]) => {
                const sc = STATUS_COLORS[status] || {}
                return (
                  <span key={status} className={styles.statusChip} style={{ background: sc.bg, color: sc.color }}>
                    {status}: {count}
                  </span>
                )
              })}
            </div>

            <div className={styles.barGroup}>
              {day.paid > 0 && (
                <div className={styles.barRow}>
                  <span className={styles.barLabel}>Paid</span>
                  <div className={styles.barTrack}>
                    <div
                      className={styles.barFill}
                      style={{
                        width: `${(day.paid / day.revenue) * 100}%`,
                        background: 'var(--success)',
                      }}
                    />
                  </div>
                  <span className={styles.barValue}>{formatCurrency(day.paid)}</span>
                </div>
              )}
              {day.pending > 0 && (
                <div className={styles.barRow}>
                  <span className={styles.barLabel}>Pending</span>
                  <div className={styles.barTrack}>
                    <div

import { useMemo } from "react";
import useStore from "../store/useStore";
import { formatCurrency } from "../utils/format";
import styles from "./DailyTotals.module.css";

export default function DailyTotals() {
  const orders = useStore((s) => s.orders);

  const dailyData = useMemo(() => {
    const map = {};
    orders.forEach((order) => {
      const date = order.createdAt.slice(0, 10);
      if (!map[date]) {
        map[date] = { date, total: 0, count: 0, paid: 0, unpaid: 0, partial: 0 };
      }
      map[date].total += order.total;
      map[date].count += 1;
      if (order.paymentStatus === "paid") map[date].paid += 1;
      else if (order.paymentStatus === "unpaid") map[date].unpaid += 1;
      else if (order.paymentStatus === "partial") map[date].partial += 1;
    });
    return Object.values(map).sort((a, b) => b.date.localeCompare(a.date));
  }, [orders]);

  const grandTotal = useMemo(
    () => dailyData.reduce((sum, d) => sum + d.total, 0),
    [dailyData]
  );

  if (dailyData.length === 0) {
    return (
      <div className={styles.empty}>
        <p>No orders to display.</p>
      </div>
    );
  }

  return (
    <div className={styles.wrapper}>
      <div className={styles.summary}>
        <div className={styles.summaryCard}>
          <span className={styles.summaryLabel}>Grand Total</span>
          <span className={styles.summaryValue}>{formatCurrency(grandTotal)}</span>
        </div>
        <div className={styles.summaryCard}>
          <span className={styles.summaryLabel}>Total Orders</span>
          <span className={styles.summaryValue}>
            {dailyData.reduce((s, d) => s + d.count, 0)}
          </span>
        </div>
        <div className={styles.summaryCard}>
          <span className={styles.summaryLabel}>Days Tracked</span>
          <span className={styles.summaryValue}>{dailyData.length}</span>
        </div>
      </div>

      <div className={styles.tableWrapper}>
        <table className={styles.table}>
          <thead>
            <tr>
              <th>Date</th>
              <th>Orders</th>
              <th className={styles.right}>Daily Total</th>
              <th className={styles.center}>Paid</th>
              <th className={styles.center}>Partial</th>
              <th className={styles.center}>Unpaid</th>
              <th className={styles.right}>Avg Order</th>
            </tr>
          </thead>
          <tbody>
            {dailyData.map((row) => (
              <tr key={row.date} className={styles.row}>
                <td className={styles.date}>
                  {new Date(row.date + "T00:00:00").toLocaleDateString("en-US", {
                    weekday: "short",
                    year: "numeric",
                    month: "short",
                    day: "numeric",
                  })}
                </td>
                <td className={styles.center}>{row.count}</td>
                <td className={`${styles.right} ${styles.total}`}>
                  {formatCurrency(row.total)}
                </td>
                <td className={styles.center}>
                  {row.paid > 0 ? (
                    <span className={`${styles.badge} ${styles.paid}`}>{row.paid}</span>
                  ) : (
                    <span className={styles.zero}>—</span>
                  )}
                </td>
                <td className={styles.center}>
                  {row.partial > 0 ? (
                    <span className={`${styles.badge} ${styles.partial}`}>{row.partial}</span>
                  ) : (
                    <span className={styles.zero}>—</span>
                  )}
                </td>
                <td className={styles.center}>
                  {row.unpaid > 0 ? (
                    <span className={`${styles.badge} ${styles.unpaid}`}>{row.unpaid}</span>
                  ) : (
                    <span className={styles.zero}>—</span>
                  )}
                </td>
                <td className={styles.right}>
                  {formatCurrency(row.total / row.count)}
                </td>
              </tr>
            ))}
          </tbody>
          <tfoot>
            <tr className={styles.footRow}>
              <td>
                <strong>Total</strong>
              </td>
              <td className={styles.center}>
                <strong>{dailyData.reduce((s, d) => s + d.count, 0)}</strong>
              </td>
              <td className={`${styles.right} ${styles.total}`}>
                <strong>{formatCurrency(grandTotal)}</strong>
              </td>
              <td className={styles.center}>
                <strong>{dailyData.reduce((s, d) => s + d.paid, 0)}</strong>
              </td>
              <td className={styles.center}>
                <strong>{dailyData.reduce((s, d) => s + d.partial, 0)}</strong>
              </td>
              <td className={styles.center}>
                <strong>{dailyData.reduce((s, d) => s + d.unpaid, 0)}</strong>
              </td>
              <td className={styles.right}>—</td>
            </tr>
          </tfoot>
        </table>
      </div>
    </div>
  );
}
