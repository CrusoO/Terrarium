import React, { useState, useMemo } from 'react'
import { INITIAL_STUDENTS } from './data/students'
import Toolbar from './components/Toolbar'
import StudentTable from './components/StudentTable'
import StudentDetail from './components/StudentDetail'
import AddStudentModal from './components/AddStudentModal'
import styles from './App.module.css'

export default function App() {
  const [students, setStudents] = useState(INITIAL_STUDENTS)
  const [search, setSearch] = useState('')
  const [statusFilter, setStatusFilter] = useState('all')
  const [gradeFilter, setGradeFilter] = useState('all')
  const [selectedStudent, setSelectedStudent] = useState(null)
  const [showAdd, setShowAdd] = useState(false)
  const [sortConfig, setSortConfig] = useState({ key: 'lastName', dir: 'asc' })

  const filtered = useMemo(() => {
    let list = students.filter(s => {
      const q = search.toLowerCase()
      const matchSearch =
        !q ||
        s.firstName.toLowerCase().includes(q) ||
        s.lastName.toLowerCase().includes(q) ||
        s.email.toLowerCase().includes(q) ||
        s.grade.toLowerCase().includes(q)
      const matchStatus = statusFilter === 'all' || s.status === statusFilter
      const matchGrade = gradeFilter === 'all' || s.grade === gradeFilter
      return matchSearch && matchStatus && matchGrade
    })

    list = [...list].sort((a, b) => {
      const av = a[sortConfig.key] ?? ''
      const bv = b[sortConfig.key] ?? ''
      const cmp = typeof av === 'number'
        ? av - bv
        : String(av).localeCompare(String(bv))
      return sortConfig.dir === 'asc' ? cmp : -cmp
    })

    return list
  }, [students, search, statusFilter, gradeFilter, sortConfig])

  function handleSort(key) {
    setSortConfig(prev =>
      prev.key === key
        ? { key, dir: prev.dir === 'asc' ? 'desc' : 'asc' }
        : { key, dir: 'asc' }
    )
  }

  function handleAddStudent(data) {
    const newStudent = {
      ...data,
      id: Date.now(),
    }
    setStudents(prev => [...prev, newStudent])
    setShowAdd(false)
  }

  function handleUpdateStudent(updated) {
    setStudents(prev => prev.map(s => s.id === updated.id ? updated : s))
    setSelectedStudent(updated)
  }

  function handleDeleteStudent(id) {
    setStudents(prev => prev.filter(s => s.id !== id))
    setSelectedStudent(null)
  }

  const counts = useMemo(() => ({
    all: students.length,
    active: students.filter(s => s.status === 'active').length,
    inactive: students.filter(s => s.status === 'inactive').length,
    probation: students.filter(s => s.status === 'probation').length,
  }), [students])

  return (
    <div className={styles.app}>
      <header className={styles.header}>
        <div className={styles.headerInner}>
          <div className={styles.headerLeft}>
            <div className={styles.logo}>
              <svg width="28" height="28" viewBox="0 0 24 24" fill="none">
                <path d="M12 3L1 9l11 6 9-4.91V17h2V9L12 3z" fill="currentColor"/>
                <path d="M5 13.18v4L12 21l7-3.82v-4L12 17l-7-3.82z" fill="currentColor" opacity=".7"/>
              </svg>
            </div>
            <div>
              <h1 className={styles.title}>Class Roster</h1>
              <p className={styles.subtitle}>Advanced Biology · Period 3 · Ms. Rivera</p>
            </div>
          </div>
          <button className={styles.addBtn} onClick={() => setShowAdd(true)}>
            <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5">
              <line x1="12" y1="5" x2="12" y2="19"/><line x1="5" y1="12" x2="19" y2="12"/>
            </svg>
            Add Student
          </button>
        </div>
      </header>

      <main className={styles.main}>
        <Toolbar
          search={search}
          onSearch={setSearch}
          statusFilter={statusFilter}
          onStatusFilter={setStatusFilter}
          gradeFilter={gradeFilter}
          onGradeFilter={setGradeFilter}
          counts={counts}
          total={filtered.length}
        />

        <div className={styles.content}>
          <StudentTable
            students={filtered}
            onSelect={setSelectedStudent}
            selectedId={selectedStudent?.id}
            sortConfig={sortConfig}
            onSort={handleSort}
          />

          {selectedStudent && (
            <StudentDetail
              student={selectedStudent}
              onClose={() => setSelectedStudent(null)}
              onUpdate={handleUpdateStudent}
              onDelete={handleDeleteStudent}
            />
          )}
        </div>
      </main>

      {showAdd && (
        <AddStudentModal
          onAdd={handleAddStudent}
          onClose={() => setShowAdd(false)}
        />
      )}
    </div>
  )
}

import React from 'react'
import StatusBadge from './StatusBadge'
import styles from './StudentTable.module.css'

const COLUMNS = [
  { key: 'lastName',  label: 'Name' },
  { key: 'grade',     label: 'Grade' },
  { key: 'gpa',       label: 'GPA' },
  { key: 'status',    label: 'Status' },
  { key: 'email',     label: 'Email' },
]

export default function StudentTable({ students, onSelect, selectedId, sortConfig, onSort }) {
  function SortIcon({ col }) {
    if (sortConfig.key !== col) return (
      <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" opacity=".3">
        <path d="M7 15l5 5 5-5M7 9l5-5 5 5"/>
      </svg>
    )
    return sortConfig.dir === 'asc'
      ? <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5"><path d="M7 15l5 5 5-5"/></svg>
      : <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5"><path d="M7 9l5-5 5 5"/></svg>
  }

  if (students.length === 0) {
    return (
      <div className={styles.empty}>
        <svg width="48" height="48" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.5" opacity=".3">
          <circle cx="11" cy="11" r="8"/><line x1="21" y1="21" x2="16.65" y2="16.65"/>
        </svg>
        <p>No students match your search</p>
        <span>Try adjusting your filters</span>
      </div>
    )
  }

  return (
    <div className={styles.wrapper}>
      <table className={styles.table}>
        <thead>
          <tr>
            {COLUMNS.map(col => (
              <th key={col.key} onClick={() => onSort(col.key)} className={styles.th}>
                <span className={styles.thInner}>
                  {col.label}
                  <SortIcon col={col.key} />
                </span>
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {students.map(student => (
            <tr
              key={student.id}
              className={`${styles.row} ${selectedId === student.id ? styles.rowSelected : ''}`}
              onClick={() => onSelect(student)}
            >
              <td className={styles.td}>
                <div className={styles.nameCell}>
                  <div className={styles.avatar} data-status={student.status}>
                    {student.firstName[0]}{student.lastName[0]}
                  </div>
                  <div>
                    <div className={styles.fullName}>{student.firstName} {student.lastName}</div>
                    <div className={styles.emailSmall}>{student.email}</div>
                  </div>
                </div>
              </td>
              <td className={styles.td}>
                <span className={styles.gradeChip}>{student.grade}</span>
              </td>
              <td className={styles.td}>
                <span className={`${styles.gpa} ${student.gpa >= 3.5 ? styles.gpaHigh : student.gpa >= 2.5 ? styles.gpaMid : styles.gpaLow}`}>
                  {student.gpa.toFixed(1)}
                </span>
              </td>
              <td className={styles.td}>
                <StatusBadge status={student.status} />
              </td>
              <td className={`${styles.td} ${styles.emailCol}`}>{student.email}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}

import React, { useState } from 'react'
import { GRADES, STATUSES } from '../data/students'
import styles from './AddStudentModal.module.css'

const EMPTY = {
  firstName: '', lastName: '', email: '', phone: '',
  grade: '9th', gpa: '', status: 'active', address: '', notes: ''
}

export default function AddStudentModal({ onAdd, onClose }) {
  const [form, setForm] = useState(EMPTY)
  const [errors, setErrors] = useState({})

  function handleChange(e) {
    const { name, value } = e.target
    setForm(prev => ({ ...prev, [name]: value }))
    if (errors[name]) setErrors(prev => ({ ...prev, [name]: '' }))
  }

  function validate() {
    const e = {}
    if (!form.firstName.trim()) e.firstName = 'Required'
    if (!form.lastName.trim()) e.lastName = 'Required'
    if (!form.email.trim()) e.email = 'Required'
    else if (!/\S+@\S+\.\S+/.test(form.email)) e.email = 'Invalid email'
    if (!form.gpa) e.gpa = 'Required'
    else if (isNaN(form.gpa) || +form.gpa < 0 || +form.gpa > 4) e.gpa = '0.0 – 4.0'
    return e
  }

  function handleSubmit(e) {
    e.preventDefault()
    const errs = validate()
    if (Object.keys(errs).length) { setErrors(errs); return }
    onAdd({ ...form, gpa: parseFloat(form.gpa) })
  }

  return (
    <div className={styles.overlay} onClick={e => e.target === e.currentTarget && onClose()}>
      <div className={styles.modal}>
        <div className={styles.modalHeader}>
          <div className={styles.modalIcon}>
            <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5">
              <path d="M16 21v-2a4 4 0 00-4-4H6a4 4 0 00-4 4v2"/>
              <circle cx="9" cy="7" r="4"/>
              <line x1="19" y1="8" x2="19" y2="14"/><line x1="22" y1="11" x2="16" y2="11"/>
            </svg>
          </div>
          <div>
            <h2 className={styles.modalTitle}>Add New Student</h2>
            <p className={styles.modalSub}>Fill in the student's information below</p>
          </div>
          <button className={styles.closeBtn} onClick={onClose} aria-label="Close">
            <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5">
              <line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/>
            </svg>
          </button>
        </div>

        <form className={styles.form} onSubmit={handleSubmit} noValidate>
          <div className={styles.section}>
            <h3 className={styles.sectionTitle}>Personal Information</h3>
            <div className={styles.grid}>
              <Field label="First Name *" error={errors.firstName}>
                <input name="firstName" value={form.firstName} onChange={handleChange} placeholder="e.g. Jane" className={errors.firstName ? styles.inputError : ''} />
              </Field>
              <Field label="Last Name *" error={errors.lastName}>
                <input name="lastName" value={form.lastName} onChange={handleChange} placeholder="e.g. Smith" className={errors.lastName ? styles.inputError : ''} />
              </Field>
              <Field label="Email Address *" error={errors.email}>
                <input name="email" type="email" value={form.email} onChange={handleChange} placeholder="jane.smith@school.edu" className={errors.email ? styles.inputError : ''} />
              </Field>
              <Field label="Phone Number">
                <input name="phone" value={form.phone} onChange={handleChange} placeholder="555-0100" />
              </Field>
              <Field label="Home Address" className={styles.fullWidth}>
                <input name="address" value={form.address} onChange={handleChange} placeholder="123 Main St" />
              </Field>
            </div>
          </div>

          <div className={styles.section}>
            <h3 className={styles.sectionTitle}>Academic Information</h3>
            <div className={styles.grid}>
              <Field label="Grade Level">
                <select name="grade" value={form.grade} onChange={handleChange}>
                  {GRADES.map(g => <option key={g} value={g}>{g} Grade</option>)}
                </select>
              </Field>
              <Field label="GPA *" error={errors.gpa}>
                <input name="gpa" type="number" step="0.1" min="0" max="4" value={form.gpa} onChange={handleChange} placeholder="0.0 – 4.0" className={errors.gpa ? styles.inputError : ''} />
              </Field>
              <Field label="Enrollment Status">
                <select name="status" value={form.status} onChange={handleChange}>
                  {STATUSES.map(s => <option key={s} value={s}>{s.charAt(0).toUpperCase() + s.slice(1)}</option>)}
                </select>
              </Field>
            </div>
          </div>

          <div className={styles.section}>
            <h3 className={styles.sectionTitle}>Notes</h3>
            <textarea
              name="notes"
              value={form.notes}
              onChange={handleChange}
              rows={3}
              placeholder="Any relevant notes about this student…"
              className={styles.textarea}
            />
          </div>

            <button type="button" className={styles.cancelBtn} onClick={onClose}>Cancel</button>

              </svg>
              Add Student
            </button>
          </div>
        </form>
      </div>
    </div>
  )
