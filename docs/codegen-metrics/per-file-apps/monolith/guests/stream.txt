import React, { useState, useMemo } from "react";
import { INITIAL_GUESTS } from "./data/guests";
import Header from "./components/Header";
import StatsBar from "./components/StatsBar";
import Toolbar from "./components/Toolbar";
import GuestGrid from "./components/GuestGrid";
import GuestDetail from "./components/GuestDetail";
import AddGuestModal from "./components/AddGuestModal";
import styles from "./App.module.css";

export default function App() {
  const [guests, setGuests] = useState(INITIAL_GUESTS);
  const [search, setSearch] = useState("");
  const [filter, setFilter] = useState("all");
  const [selectedGuest, setSelectedGuest] = useState(null);
  const [showAdd, setShowAdd] = useState(false);

  const filtered = useMemo(() => {
    return guests.filter((g) => {
      const matchSearch =
        g.name.toLowerCase().includes(search.toLowerCase()) ||
        g.email.toLowerCase().includes(search.toLowerCase());
      const matchFilter = filter === "all" || g.rsvp === filter;
      return matchSearch && matchFilter;
    });
  }, [guests, search, filter]);

  const stats = useMemo(() => ({
    total: guests.length,
    yes: guests.filter((g) => g.rsvp === "yes").length,
    no: guests.filter((g) => g.rsvp === "no").length,
    maybe: guests.filter((g) => g.rsvp === "maybe").length,
    pending: guests.filter((g) => g.rsvp === "pending").length,
  }), [guests]);

  function addGuest(data) {
    const newGuest = {
      ...data,
      id: Date.now(),
      avatar: data.name.split(" ").map((w) => w[0]).join("").slice(0, 2).toUpperCase(),
    };
    setGuests((prev) => [newGuest, ...prev]);
    setShowAdd(false);
  }

  function updateGuest(updated) {
    setGuests((prev) => prev.map((g) => (g.id === updated.id ? updated : g)));
    setSelectedGuest(updated);
  }

  function deleteGuest(id) {
    setGuests((prev) => prev.filter((g) => g.id !== id));
    setSelectedGuest(null);
  }

  return (
    <div className={styles.app}>
      <Header />
      <main className={styles.main}>
        <StatsBar stats={stats} activeFilter={filter} onFilter={setFilter} />
        <Toolbar
          search={search}
          onSearch={setSearch}
          onAdd={() => setShowAdd(true)}
          count={filtered.length}
        />
        <GuestGrid
          guests={filtered}
          onSelect={setSelectedGuest}
          selectedId={selectedGuest?.id}
        />
        {filtered.length === 0 && (
          <div className={styles.empty}>
            <span className={styles.emptyIcon}>🔍</span>
            <p>No guests match your search.</p>
          </div>
        )}
      </main>

      {selectedGuest && (
        <GuestDetail
          guest={selectedGuest}
          onClose={() => setSelectedGuest(null)}
          onUpdate={updateGuest}
          onDelete={deleteGuest}
        />
      )}

      {showAdd && (
        <AddGuestModal onClose={() => setShowAdd(false)} onAdd={addGuest} />
      )}
    </div>
  );
}

import React, { useState } from "react";
import styles from "./GuestDetail.module.css";
import { rsvpMeta, RSVP_OPTIONS } from "../utils/rsvp";

export default function GuestDetail({ guest, onClose, onUpdate, onDelete }) {
  const [editing, setEditing] = useState(false);
  const [form, setForm] = useState({ ...guest });
  const meta = rsvpMeta(guest.rsvp);

  function handleChange(e) {
    const { name, value } = e.target;
    setForm((f) => ({ ...f, [name]: value }));
  }

  function handleSave() {
    onUpdate(form);
    setEditing(false);
  }

  function handleCancel() {
    setForm({ ...guest });
    setEditing(false);
  }

  return (
    <div className={styles.overlay} onClick={onClose}>
      <aside className={styles.panel} onClick={(e) => e.stopPropagation()}>
        <div className={styles.panelHeader}>
          <h2 className={styles.panelTitle}>Guest Details</h2>
          <button className={styles.closeBtn} onClick={onClose}>✕</button>
        </div>

        <div className={styles.hero}>
          <div className={styles.bigAvatar}>{guest.avatar}</div>
          {!editing ? (
            <>
              <h3 className={styles.heroName}>{guest.name}</h3>
              <span className={styles.heroBadge} style={{ "--rsvp-color": meta.color }}>
                {meta.icon} {meta.label}
              </span>
            </>
          ) : (
            <input
              className={styles.editInput}
              name="name"
              value={form.name}
              onChange={handleChange}
              placeholder="Full name"
            />
          )}
        </div>

        <div className={styles.fields}>
          <Field label="Email" icon="✉️">
            {editing
              ? <input className={styles.editInput} name="email" value={form.email} onChange={handleChange} />
              : <span>{guest.email}</span>}
          </Field>
          <Field label="Phone" icon="📞">
            {editing
              ? <input className={styles.editInput} name="phone" value={form.phone} onChange={handleChange} />
              : <span>{guest.phone}</span>}
          </Field>
          <Field label="RSVP Status" icon="📋">
            {editing
              ? (
                <select className={styles.editSelect} name="rsvp" value={form.rsvp} onChange={handleChange}>
                  {RSVP_OPTIONS.map((o) => (
                    <option key={o.key} value={o.key}>{o.label}</option>
                  ))}
                </select>
              )
              : <span style={{ color: meta.color }}>{meta.icon} {meta.label}</span>}
          </Field>
          <Field label="Meal Preference" icon="🍽">
            {editing
              ? <input className={styles.editInput} name="meal" value={form.meal} onChange={handleChange} />
              : <span>{guest.meal}</span>}
          </Field>
          <Field label="Table" icon="🪑">
            {editing
              ? <input className={styles.editInput} name="table" type="number" value={form.table || ""} onChange={handleChange} placeholder="—" />
              : <span>{guest.table ? `Table ${guest.table}` : "—"}</span>}
          </Field>
          <Field label="Notes" icon="📝">
            {editing
              ? <textarea className={styles.editTextarea} name="notes" value={form.notes} onChange={handleChange} rows={3} />
              : <span className={styles.notes}>{guest.notes || "—"}</span>}
          </Field>
        </div>

        <div className={styles.actions}>
          {editing ? (
            <>
              <button className={styles.saveBtn} onClick={handleSave}>Save Changes</button>
              <button className={styles.cancelBtn} onClick={handleCancel}>Cancel</button>
            </>
          ) : (
            <>
              <button className={styles.editBtn} onClick={() => setEditing(true)}>✏️ Edit</button>
              <button className={styles.deleteBtn} onClick={() => onDelete(guest.id)}>🗑 Remove</button>
            </>
          )}
        </div>
      </aside>
    </div>
  );
}

function Field({ label, icon, children }) {
  return (
    <div className={styles.field}>
      <span className={styles.fieldLabel}>{icon} {label}</span>
      <div className={styles.fieldValue}>{children}</div>
    </div>
  );
}

import React, { useState } from "react";
import styles from "./AddGuestModal.module.css";
import { RSVP_OPTIONS } from "../utils/rsvp";

const EMPTY = {
  name: "", email: "", phone: "", rsvp: "pending",
  meal: "Standard", table: "", notes: "",
};

export default function AddGuestModal({ onClose, onAdd }) {
  const [form, setForm] = useState(EMPTY);
  const [errors, setErrors] = useState({});

  function handleChange(e) {
    const { name, value } = e.target;
    setForm((f) => ({ ...f, [name]: value }));
    if (errors[name]) setErrors((e) => ({ ...e, [name]: "" }));
  }

  function validate() {
    const errs = {};
    if (!form.name.trim()) errs.name = "Name is required";
    if (!form.email.trim()) errs.email = "Email is required";
    else if (!/\S+@\S+\.\S+/.test(form.email)) errs.email = "Invalid email";
    return errs;
  }

  function handleSubmit(e) {
    e.preventDefault();
    const errs = validate();
    if (Object.keys(errs).length) { setErrors(errs); return; }
    onAdd({ ...form, table: form.table ? Number(form.table) : null });
  }

  return (
    <div className={styles.overlay} onClick={onClose}>
      <div className={styles.modal} onClick={(e) => e.stopPropagation()}>
        <div className={styles.modalHeader}>
          <h2 className={styles.modalTitle}>Add New Guest</h2>
          <button className={styles.closeBtn} onClick={onClose}>✕</button>
        </div>

        <form className={styles.form} onSubmit={handleSubmit}>
          <FormField label="Full Name *" error={errors.name}>
            <input
              className={`${styles.input} ${errors.name ? styles.inputError : ""}`}
              name="name" value={form.name} onChange={handleChange}
              placeholder="Jane Doe"
            />
          </FormField>

          <FormField label="Email *" error={errors.email}>
            <input
              className={`${styles.input} ${errors.email ? styles.inputError : ""}`}
              name="email" type="email" value={form.email} onChange={handleChange}
              placeholder="jane@example.com"
            />
          </FormField>

          <FormField label="Phone">
            <input
              className={styles.input}
              name="phone" value={form.phone} onChange={handleChange}
              placeholder="+1 555-0000"
            />
          </FormField>

          <div className={styles.row}>
            <FormField label="RSVP Status">
              <select className={styles.input} name="rsvp" value={form.rsvp} onChange={handleChange}>
                {RSVP_OPTIONS.map((o) => (
                  <option key={o.key} value={o.key}>{o.label}</option>
                ))}
              </select>
            </FormField>

            <FormField label="Table #">
              <input
                className={styles.input}
                name="table" type="number" value={form.table} onChange={handleChange}
                placeholder="—" min="1"
              />
            </FormField>
          </div>

          <FormField label="Meal Preference">
            <select className={styles.input} name="meal" value={form.meal} onChange={handleChange}>
              {["Standard", "Vegetarian", "Vegan", "Gluten-Free", "Halal", "Kosher"].map((m) => (
                <option key={m}>{m}</option>
              ))}
            </select>
          </FormField>

          <FormField label="Notes">
            <textarea
              className={styles.input}
              name="notes" value={form.notes} onChange={handleChange}
              placeholder="Any special requirements…"
              rows={3}
              style={{ resize: "vertical" }}
            />
          </FormField>

          <div className={styles.formActions}>
            <button type="button" className={styles.cancelBtn} onClick={onClose}>Cancel</button>
            <button type="submit" className={styles.submitBtn}>Add Guest</button>
          </div>
        </form>
      </div>
    </div>
  );
}

function FormField({ label, error, children }) {
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
      <label style={{ fontSize: "0.78rem", fontWeight: 600, color: "var(--muted)", textTransform: "uppercase", letterSpacing: "0.05em" }}>
        {label}
      </label>
      {children}
      {error && <span style={{ fontSize: "0.75rem", color: "var(--no)" }}>{error}</span>}
    </div>
  );
}

export const RSVP_OPTIONS = [
  { key: "yes", label: "Attending", icon: "✅" },
  { key: "no", label: "Declined", icon: "❌" },
  { key: "maybe", label: "Maybe", icon: "🤔" },
  { key: "pending", label: "Pending", icon: "⏳" },
];

const COLOR_MAP = {
  yes: "var(--yes)",
  no: "var(--no)",
  maybe: "var(--maybe)",
  pending: "var(--pending)",
};

export function rsvpMeta(rsvp) {
  const opt = RSVP_OPTIONS.find((o) => o.key === rsvp) || RSVP_OPTIONS[3];
  return { ...opt, color: COLOR_MAP[rsvp] || COLOR_MAP.pending };
}
