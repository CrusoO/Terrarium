import { create } from 'zustand'
import { seedVisits } from '../data/seed'

const load = () => {
  try {
    const s = localStorage.getItem('appt-visits')
    return s ? JSON.parse(s) : seedVisits
  } catch { return seedVisits }
}

const save = visits => {
  try { localStorage.setItem('appt-visits', JSON.stringify(visits)) } catch {}
}

export const useStore = create((set, get) => ({
  visits: load(),
  addVisit: visit => {
    const next = [visit, ...get().visits]
    save(next)
    set({ visits: next })
  },
  updateVisit: updated => {
    const next = get().visits.map(v => v.id === updated.id ? updated : v)
    save(next)
    set({ visits: next })
  },
  deleteVisit: id => {
    const next = get().visits.filter(v => v.id !== id)
    save(next)
    set({ visits: next })
  },
}))

import React, { useState, useMemo } from 'react'
import { useStore } from '../store/useStore'
import { STATUSES, STATUS_COLOR } from '../data/seed'
import { format, isToday, isTomorrow, isYesterday, parseISO } from 'date-fns'
import StatusBadge from '../components/StatusBadge'
import styles from './ListView.module.css'

const ALL = 'All'

function friendlyDate(iso) {
  const d = parseISO(iso)
  const time = format(d, 'h:mm a')
  if (isToday(d)) return `Today, ${time}`
  if (isTomorrow(d)) return `Tomorrow, ${time}`
  if (isYesterday(d)) return `Yesterday, ${time}`
  return format(d, 'MMM d, yyyy · h:mm a')
}

export default function ListView({ onSelect, onNew }) {
  const visits = useStore(s => s.visits)
  const [search, setSearch] = useState('')
  const [statusFilter, setStatusFilter] = useState(ALL)
  const [sort, setSort] = useState('dateAsc')

  const filtered = useMemo(() => {
    let list = [...visits]
    if (statusFilter !== ALL) list = list.filter(v => v.status === statusFilter)
    if (search.trim()) {
      const q = search.toLowerCase()
      list = list.filter(v =>
        v.patientName.toLowerCase().includes(q) ||
        v.providerName.toLowerCase().includes(q) ||
        v.visitType.toLowerCase().includes(q) ||
        v.patientPhone.includes(q)
      )
    }
    list.sort((a, b) => {
      if (sort === 'dateAsc')  return new Date(a.dateTime) - new Date(b.dateTime)
      if (sort === 'dateDesc') return new Date(b.dateTime) - new Date(a.dateTime)
      if (sort === 'patient')  return a.patientName.localeCompare(b.patientName)
      if (sort === 'provider') return a.providerName.localeCompare(b.providerName)
      return 0
    })
    return list
  }, [visits, search, statusFilter, sort])

  const counts = useMemo(() => {
    const c = { All: visits.length }
    STATUSES.forEach(s => { c[s] = visits.filter(v => v.status === s).length })
    return c
  }, [visits])

  return (
    <div className={styles.container}>
      <div className={styles.topBar}>
        <div className={styles.searchWrap}>
          <span className={styles.searchIcon}>🔍</span>
          <input
            className={styles.search}
            placeholder="Search patient, provider, visit type…"
            value={search}
            onChange={e => setSearch(e.target.value)}
          />
          {search && (
            <button className={styles.clearBtn} onClick={() => setSearch('')}>✕</button>
          )}
        </div>
        <select className={styles.sortSelect} value={sort} onChange={e => setSort(e.target.value)}>
          <option value="dateAsc">Date ↑</option>
          <option value="dateDesc">Date ↓</option>
          <option value="patient">Patient A–Z</option>
          <option value="provider">Provider A–Z</option>
        </select>
      </div>

      <div className={styles.filterRow}>
        {[ALL, ...STATUSES].map(s => (
          <button
            key={s}
            className={`${styles.filterChip} ${statusFilter === s ? styles.active : ''}`}
            style={statusFilter === s && s !== ALL ? {
              background: STATUS_COLOR[s]?.bg,
              color: STATUS_COLOR[s]?.text,
              borderColor: STATUS_COLOR[s]?.text + '55',
            } : {}}
            onClick={() => setStatusFilter(s)}
          >
            {s}
            <span className={styles.chipCount}>{counts[s] ?? 0}</span>
          </button>
        ))}
      </div>

      <div className={styles.resultMeta}>
        {filtered.length} visit{filtered.length !== 1 ? 's' : ''}
        {statusFilter !== ALL && ` · ${statusFilter}`}
        {search && ` · "${search}"`}
      </div>

      {filtered.length === 0 ? (
        <div className={styles.empty}>
          <div className={styles.emptyIcon}>🗓️</div>
          <p>No visits found</p>
          <button className={styles.emptyBtn} onClick={onNew}>Create a visit</button>
        </div>
      ) : (
        <div className={styles.list}>
          {filtered.map(v => (
            <button key={v.id} className={styles.card} onClick={() => onSelect(v.id)}>
              <div className={styles.cardLeft}>
                <div className={styles.dateBlock}>
                  <span className={styles.dateDay}>{format(parseISO(v.dateTime), 'd')}</span>
                  <span className={styles.dateMon}>{format(parseISO(v.dateTime), 'MMM')}</span>
                </div>
              </div>
              <div className={styles.cardBody}>
                <div className={styles.cardTop}>
                  <span className={styles.patientName}>{v.patientName}</span>
                  <StatusBadge status={v.status} />
                </div>
                <div className={styles.cardMeta}>
                  <span className={styles.metaItem}>🕐 {friendlyDate(v.dateTime)}</span>
                  <span className={styles.metaItem}>👨‍⚕️ {v.providerName}</span>
                  <span className={styles.metaItem}>📋 {v.visitType}</span>
                </div>
                {v.notes && <p className={styles.cardNotes}>{v.notes}</p>}
              </div>
              <span className={styles.chevron}>›</span>
            </button>
          ))}
        </div>
      )}
    </div>
  )
}

import React, { useState } from 'react'
import { useStore } from '../store/useStore'
import { STATUSES, STATUS_COLOR, PROVIDERS, VISIT_TYPES } from '../data/seed'
import { format, parseISO } from 'date-fns'
import StatusBadge from '../components/StatusBadge'
import styles from './DetailView.module.css'

export default function DetailView({ id, onBack }) {
  const visits = useStore(s => s.visits)
  const updateVisit = useStore(s => s.updateVisit)
  const deleteVisit = useStore(s => s.deleteVisit)

  const visit = visits.find(v => v.id === id)
  const [editing, setEditing] = useState(false)
  const [form, setForm] = useState(null)
  const [confirmDelete, setConfirmDelete] = useState(false)

  if (!visit) return (
    <div className={styles.notFound}>
      <p>Visit not found.</p>
      <button onClick={onBack}>← Back</button>
    </div>
  )

  const startEdit = () => {
    setForm({
      patientName: visit.patientName,
      patientDob: visit.patientDob,
      patientPhone: visit.patientPhone,
      patientEmail: visit.patientEmail,
      providerId: visit.providerId,
      visitType: visit.visitType,
      status: visit.status,
      dateTime: visit.dateTime.slice(0, 16),
      duration: visit.duration,
      notes: visit.notes,
    })
    setEditing(true)
  }

  const cancelEdit = () => { setEditing(false); setForm(null) }

  const saveEdit = () => {
    const provider = PROVIDERS.find(p => p.id === form.providerId)
    updateVisit({
      ...visit,
      ...form,
      providerName: provider.name,
      providerSpecialty: provider.specialty,
      dateTime: new Date(form.dateTime).toISOString(),
      duration: Number(form.duration),
    })
    setEditing(false)
    setForm(null)
  }

  const handleDelete = () => {
    deleteVisit(id)
    onBack()
  }

  const f = form || {}

  return (
    <div className={styles.container}>
      <div className={styles.breadcrumb}>
        <button className={styles.backBtn} onClick={onBack}>← All Visits</button>
        <span className={styles.sep}>/</span>
        <span className={styles.crumb}>{visit.patientName}</span>
      </div>

      <div className={styles.card}>
        <div className={styles.cardHeader}>
          <div className={styles.headerLeft}>
            <div className={styles.avatar}>{visit.patientName.split(' ').map(n => n[0]).join('').slice(0,2)}</div>
            <div>
              <h1 className={styles.patientName}>{visit.patientName}</h1>
              <p className={styles.subLine}>
                DOB: {visit.patientDob} &nbsp;·&nbsp; {visit.visitType}
              </p>
            </div>
          </div>

            <StatusBadge status={visit.status} large />
            {!editing && (
              <div className={styles.headerActions}>
                <button className={styles.editBtn} onClick={startEdit}>✏️ Edit</button>
                <button className={styles.deleteBtn} onClick={() => setConfirmDelete(true)}>🗑️ Delete</button>
              </div>
            )}
          </div>
        </div>

        {editing ? (
          <EditForm form={f} setForm={setForm} onSave={saveEdit} onCancel={cancelEdit} />
        ) : (
          <ReadView visit={visit} />
        )}
      </div>

      {confirmDelete && (
        <div className={styles.overlay}>
          <div className={styles.dialog}>
            <h3>Delete this visit?</h3>
            <p>This action cannot be undone.</p>
            <div className={styles.dialogActions}>
              <button className={styles.cancelBtn} onClick={() => setConfirmDelete(false)}>Cancel</button>
              <button className={styles.confirmDeleteBtn} onClick={handleDelete}>Delete</button>
            </div>
          </div>
        </div>
      )}
    </div>
  )
}

import React, { useState } from 'react'
import { v4 as uuid } from 'uuid'
import { STATUSES, PROVIDERS, VISIT_TYPES } from '../data/seed'
import { useStore } from '../store/useStore'
import styles from './CreateModal.module.css'

const defaultForm = () => {
  const now = new Date()
  now.setMinutes(Math.ceil(now.getMinutes() / 15) * 15, 0, 0)
  return {
    patientName: '',
    patientDob: '',
    patientPhone: '',
    patientEmail: '',
    providerId: PROVIDERS[0].id,
    visitType: VISIT_TYPES[0],
    status: 'Scheduled',
    dateTime: now.toISOString().slice(0, 16),
    duration: 30,
    notes: '',
  }
}

export default function CreateModal({ onClose, onCreated }) {
  const addVisit = useStore(s => s.addVisit)
  const [form, setForm] = useState(defaultForm)
  const [errors, setErrors] = useState({})

  const set = (k, v) => {
    setForm(f => ({ ...f, [k]: v }))
    setErrors(e => ({ ...e, [k]: undefined }))
  }

  const validate = () => {
    const e = {}
    if (!form.patientName.trim()) e.patientName = 'Required'
    if (!form.dateTime) e.dateTime = 'Required'
    return e
  }

  const handleSubmit = e => {
    e.preventDefault()
    const errs = validate()
    if (Object.keys(errs).length) { setErrors(errs); return }
    const provider = PROVIDERS.find(p => p.id === form.providerId)
    const visit = {
      id: uuid(),
      ...form,
      providerName: provider.name,
      providerSpecialty: provider.specialty,
      dateTime: new Date(form.dateTime).toISOString(),
      duration: Number(form.duration),
      createdAt: new Date().toISOString(),
    }
    addVisit(visit)
    onCreated(visit.id)
  }

  return (
    <div className={styles.overlay} onClick={e => e.target === e.currentTarget && onClose()}>
      <div className={styles.modal}>
        <div className={styles.modalHeader}>
          <h2>New Visit</h2>
          <button className={styles.closeBtn} onClick={onClose}>✕</button>
        </div>

            <h3 className={styles.sectionTitle}>Patient</h3>

              <label className={styles.group}>
                <span>Full Name <em>*</em></span>

                  value={form.patientName}
                  onChange={e => set('patientName', e.target.value)}
                  placeholder="Jane Doe"
                  className={errors.patientName ? styles.error : ''}
                />
                {errors.patientName && <span className={styles.errMsg}>{errors.patientName}</span>}
              </label>
              <label className={styles.group}>
                <span>Date of Birth</span>
                <input type="date" value={form.patientDob} onChange={e => set('patientDob', e.target.value)} />
              </label>

                <span>Phone</span>

              </label>
              <label className={styles.group}>
                <span>Email</span>
                <input type="email" value={form.patientEmail} onChange={e => set('patientEmail', e.target.value)} placeholder="patient@example.com" />
              </label>
            </div>
          </div>

          <div className={styles.section}>
            <h3 className={styles.sectionTitle}>Appointment</h3>
            <div className={styles.grid}>
              <label className={styles.group}>
                <span>Provider</span>
                <select value={form.providerId} onChange={e => set('providerId', e.target.value)}>
                  {PROVIDERS.map(p => <option key={p.id} value={p.id}>{p.name} — {p.specialty}</option>)}
                </select>
              </label>
              <label className={styles.group}>
                <span>Visit Type</span>
                <select value={form.visitType} onChange={e => set('visitType', e.target.value)}>
                  {VISIT_TYPES.map(t => <option key={t}>{t}</option>)}
                </select>
              </label>
              <label className={styles.group}>
                <span>Status</span>
                <select value={form.status} onChange={e => set('status', e.target.value)}>

                </select>
              </label>
              <label className={styles.group}>
                <span>Date & Time <em>*</em></span>
